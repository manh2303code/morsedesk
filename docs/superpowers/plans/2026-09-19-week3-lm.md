```python
"""Bảng chữ của mô hình ngôn ngữ ký tự cho một profile."""

from __future__ import annotations

import json
from pathlib import Path

from morsedesk.lm.profile import Profile

BOS = "<s>"
EOS = "</s>"
UNK = "<unk>"
SPACE_CHAR = " "
SPACE_TOKEN = "▁"  # U+2581, thay khoảng trắng trong file token hóa cho lmplz


class LMAlphabet:
    def __init__(self, symbols: list[str]) -> None:
        if symbols[:4] != [BOS, EOS, UNK, SPACE_CHAR]:
            raise ValueError("bốn ký hiệu đầu phải là <s>, </s>, <unk>, ' '")
        if len(set(symbols)) != len(symbols):
            raise ValueError("bảng chữ có ký hiệu trùng")
        self.symbols = list(symbols)
        self.index = {s: i for i, s in enumerate(self.symbols)}
        self.bos_id, self.eos_id, self.unk_id, self.space_id = 0, 1, 2, 3

    @classmethod
    def for_profile(cls, profile: Profile) -> LMAlphabet:
        letters: list[str] = []
        for options in profile.map.values():
            for ch in options:
                if ch not in letters:
                    letters.append(ch)
        return cls([BOS, EOS, UNK, SPACE_CHAR] + letters)

    def __len__(self) -> int:
        return len(self.symbols)

    def encode(self, text: str) -> list[int]:
        return [self.index.get(ch, self.unk_id) for ch in text]

    def decode(self, ids) -> str:
        return "".join(self.symbols[int(i)] for i in ids)

    def to_tokens_line(self, text: str) -> str:
        return " ".join(SPACE_TOKEN if ch == SPACE_CHAR else ch for ch in text)

    def from_tokens_line(self, line: str) -> str:
        return "".join(SPACE_CHAR if tok == SPACE_TOKEN else tok for tok in line.split())

    def save(self, path: Path) -> None:
        Path(path).write_text(json.dumps({"symbols": self.symbols}, ensure_ascii=False), encoding="utf-8")

    @classmethod
    def load(cls, path: Path) -> LMAlphabet:
        return cls(json.loads(Path(path).read_text(encoding="utf-8"))["symbols"])
```

```python
"""LM ký tự n-gram kiểu ARPA (backoff) lưu bảng băm numpy; huấn luyện nhỏ thuần Python."""

from __future__ import annotations

import math
from collections import Counter, defaultdict
from collections.abc import Sequence
from dataclasses import dataclass
from functools import lru_cache
from pathlib import Path

import numpy as np

from morsedesk.lm.alphabet import SPACE_TOKEN, LMAlphabet

FNV_OFFSET = np.uint64(0xCBF29CE484222325)
FNV_PRIME = np.uint64(0x100000001B3)
LN10 = math.log(10.0)


def _fnv_step(h: np.uint64, byte: int) -> np.uint64:
    with np.errstate(over="ignore"):
        return (h ^ np.uint64(byte)) * FNV_PRIME


def fnv1a64(ids: Sequence[int]) -> int:
    h = FNV_OFFSET
    for i in ids:
        h = _fnv_step(h, int(i) & 0xFF)
        h = _fnv_step(h, (int(i) >> 8) & 0xFF)
    return int(h)


def extend_hash(h: np.uint64, ids: np.ndarray) -> np.ndarray:
    """Hash của (prefix có hash h) + mỗi id trong ids, vector hóa."""
    ids = np.asarray(ids, dtype=np.uint64)
    with np.errstate(over="ignore"):
        x = (np.uint64(h) ^ (ids & np.uint64(0xFF))) * FNV_PRIME
        x = (x ^ ((ids >> np.uint64(8)) & np.uint64(0xFF))) * FNV_PRIME
    return x


@dataclass
class NGramTable:
    hashes: np.ndarray
    logp: np.ndarray
    backoff: np.ndarray

    def lookup(self, key: int) -> int:
        i = int(np.searchsorted(self.hashes, np.uint64(key)))
        return i if i < len(self.hashes) and int(self.hashes[i]) == key else -1

    def lookup_many(self, keys: np.ndarray) -> np.ndarray:
        idx = np.searchsorted(self.hashes, keys)
        idx_c = np.minimum(idx, max(0, len(self.hashes) - 1))
        hit = (idx < len(self.hashes)) & (self.hashes[idx_c] == keys) if len(self.hashes) else np.zeros(len(keys), bool)
        return np.where(hit, idx_c, -1)


class CharNGramLM:
    def __init__(self, alphabet: LMAlphabet, order: int, tables: list[NGramTable]) -> None:
        if len(tables) != order:
            raise ValueError("số bảng phải bằng bậc")
        self.alphabet = alphabet
        self.order = order
        self.tables = tables
        self._cache: dict[tuple[tuple[int, ...], int], float] = {}

    # ---- chấm điểm ----
    def _backoff_weight(self, context: tuple[int, ...]) -> float:
        """Tổng backoff khi lùi từ context (bậc len(context)) xuống context[1:]."""
        k = len(context)
        if k == 0:
            return 0.0
        i = self.tables[k - 1].lookup(fnv1a64(context))
        return float(self.tables[k - 1].backoff[i]) if i >= 0 else 0.0

    def score(self, context: tuple[int, ...], symbol: int) -> float:
        ctx = tuple(context)[-(self.order - 1):] if self.order > 1 else ()
        key = (ctx, symbol)
        hit = self._cache.get(key)
        if hit is not None:
            return hit
        acc = 0.0
        while True:
            k = len(ctx) + 1
            i = self.tables[k - 1].lookup(fnv1a64(ctx + (symbol,)))
            if i >= 0:
                val = acc + float(self.tables[k - 1].logp[i])
                break
            if not ctx:
                val = acc + math.log(1e-7)  # ký hiệu chưa từng thấy ở bậc 1
                break
            acc += self._backoff_weight(ctx)
            ctx = ctx[1:]
        if len(self._cache) > 100_000:
            self._cache.clear()
        self._cache[key] = val
        return val

    def score_many(self, context: tuple[int, ...], symbols: np.ndarray) -> np.ndarray:
        symbols = np.asarray(symbols, dtype=np.int64)
        out = np.empty(len(symbols), dtype=np.float32)
        pending = np.ones(len(symbols), dtype=bool)
        acc = 0.0
        ctx = tuple(context)[-(self.order - 1):] if self.order > 1 else ()
        while True:
            k = len(ctx) + 1
            base = np.uint64(fnv1a64(ctx)) if ctx else FNV_OFFSET
            keys = extend_hash(base, symbols)
            idx = self.tables[k - 1].lookup_many(keys)
            found = pending & (idx >= 0)
            out[found] = acc + self.tables[k - 1].logp[idx[found]]
            pending &= ~found
            if not pending.any():
                break
            if not ctx:
                out[pending] = acc + math.log(1e-7)
                break
            acc += self._backoff_weight(ctx)
            ctx = ctx[1:]
        return out

    def sentence_logprob(self, text: str) -> float:
        ids = [self.alphabet.bos_id] + self.alphabet.encode(text) + [self.alphabet.eos_id]
        total = 0.0
        for i in range(1, len(ids)):
            total += self.score(tuple(ids[max(0, i - self.order + 1):i]), ids[i])
        return total

    def perplexity(self, lines: list[str]) -> float:
        total_lp = 0.0
        n = 0
        for ln in lines:
            total_lp += self.sentence_logprob(ln)
            n += len(ln) + 1
        return math.exp(-total_lp / max(1, n))


# ---- huấn luyện nhỏ: Kneser-Ney nội suy đơn giản (absolute discounting) ----

def _ngram_counts(lines: list[str], alphabet: LMAlphabet, order: int) -> list[Counter]:
    counts: list[Counter] = [Counter() for _ in range(order)]
    for ln in lines:
        ids = [alphabet.bos_id] + alphabet.encode(ln) + [alphabet.eos_id]
        for i in range(1, len(ids)):
            for k in range(1, order + 1):
                if i - k + 1 < 0:
                    break
                counts[k - 1][tuple(ids[i - k + 1:i + 1])] += 1
    return counts


def train_small(lines: list[str], alphabet: LMAlphabet, order: int = 6, discount: float = 0.75) -> CharNGramLM:
    """Absolute discounting nội suy, xuất dạng backoff kiểu ARPA (entry đã nội suy, backoff = λ(h))."""
    lines = [ln for ln in lines if ln.strip()]
    counts = _ngram_counts(lines, alphabet, order)
    vocab_ids = list(range(1, len(alphabet)))  # không sinh <s>
    v = len(vocab_ids)
    probs: list[dict[tuple[int, ...], float]] = []
    backoffs: list[dict[tuple[int, ...], float]] = []

    # bậc 1
    total1 = sum(counts[0].values())
    n1plus = len(counts[0])
    lam1 = discount * n1plus / max(1, total1)
    p1 = {}
    for s in vocab_ids:
        c = counts[0].get((s,), 0)
        p1[(s,)] = max(c - discount, 0.0) / max(1, total1) + lam1 / v
    probs.append(p1)
    backoffs.append({})

    for k in range(2, order + 1):
        ctx_total: dict[tuple[int, ...], int] = defaultdict(int)
        ctx_types: dict[tuple[int, ...], int] = defaultdict(int)
        for ng, c in counts[k - 1].items():
            ctx_total[ng[:-1]] += c
            ctx_types[ng[:-1]] += 1
        pk: dict[tuple[int, ...], float] = {}
        bk: dict[tuple[int, ...], float] = {}
        lower = probs[k - 2]
        for ng, c in counts[k - 1].items():
            h = ng[:-1]
            lam = discount * ctx_types[h] / ctx_total[h]
            pk[ng] = max(c - discount, 0.0) / ctx_total[h] + lam * lower.get(ng[1:], 1.0 / v)
            bk[h] = lam
        probs.append(pk)
        backoffs.append(bk)

    tables = []
    ngrams_named: list[dict[tuple[int, ...], tuple[float, float]]] = []
    for k in range(1, order + 1):
        bo_map = backoffs[k] if k < order else {}
        named = {ng: (math.log(p), math.log(bo_map[ng]) if ng in bo_map else 0.0)
                 for ng, p in probs[k - 1].items()}
        ngrams_named.append(named)
        items = sorted(((fnv1a64(ng), lp, bo) for ng, (lp, bo) in named.items()), key=lambda t: t[0])
        tables.append(NGramTable(
            np.array([h for h, _, _ in items], dtype=np.uint64),
            np.array([lp for _, lp, _ in items], dtype=np.float32),
            np.array([bo for _, _, bo in items], dtype=np.float32),
        ))
    lm = CharNGramLM(alphabet, order, tables)
    lm._ngrams = ngrams_named  # tên n-gram, chỉ để save_arpa
    return lm


# ---- ARPA và nhị phân ----

def save_arpa(lm: CharNGramLM, path: Path) -> None:
    """Chỉ dùng cho LM nhỏ (cần tên n-gram): tái dựng từ bảng bằng cách liệt kê mọi tổ hợp là không khả thi,
    nên save_arpa yêu cầu lm có thuộc tính _ngrams (do train_small gán)."""
    ngrams = getattr(lm, "_ngrams", None)
    if ngrams is None:
        raise ValueError("save_arpa chỉ hỗ trợ LM từ train_small")
    with open(path, "w", encoding="utf-8") as f:
        f.write("\\data\\\n")
        for k in range(1, lm.order + 1):
            f.write(f"ngram {k}={len(ngrams[k - 1])}\n")
        for k in range(1, lm.order + 1):
            f.write(f"\n\\{k}-grams:\n")
            for ng, (lp, bo) in ngrams[k - 1].items():
                words = " ".join(SPACE_TOKEN if s == lm.alphabet.space_id else lm.alphabet.symbols[s] for s in ng)
                line = f"{lp / LN10:.6f}\t{words}"
                if k < lm.order and bo != 0.0:
                    line += f"\t{bo / LN10:.6f}"
                f.write(line + "\n")
        f.write("\n\\end\\\n")


def load_arpa(path: Path, alphabet: LMAlphabet) -> CharNGramLM:
    order = 0
    per_order: dict[int, list[tuple[int, float, float]]] = defaultdict(list)
    section = 0
    with open(path, encoding="utf-8") as f:
        for raw in f:
            line = raw.strip()
            if not line:
                continue
            if line.startswith("ngram "):
                order = max(order, int(line[6:].split("=")[0]))
                continue
            if line.endswith("-grams:"):
                section = int(line[1:].split("-")[0])
                continue
            if line in ("\\data\\", "\\end\\") or section == 0:
                continue
            parts = line.split("\t")
            lp = float(parts[0]) * LN10
            toks = parts[1].split(" ")
            ids = [alphabet.index.get(SPACE_TOKEN if t == SPACE_TOKEN else t, None) for t in toks]
            ids = [alphabet.space_id if t == SPACE_TOKEN else i for t, i in zip(toks, ids, strict=True)]
            if any(i is None for i in ids):
                continue  # ký hiệu ngoài bảng chữ (ví dụ <unk> của lmplz): bỏ
            bo = float(parts[2]) * LN10 if len(parts) > 2 else 0.0
            per_order[section].append((fnv1a64(ids), lp, bo))
    tables = []
    for k in range(1, order + 1):
        items = sorted(per_order.get(k, []), key=lambda t: t[0])
        tables.append(NGramTable(
            np.array([h for h, _, _ in items], dtype=np.uint64),
            np.array([lp for _, lp, _ in items], dtype=np.float32),
            np.array([bo for _, _, bo in items], dtype=np.float32),
        ))
    return CharNGramLM(alphabet, order, tables)


def save_bin(lm: CharNGramLM, path: Path) -> None:
    """Nhị phân numpy; bảng chữ lưu dạng chuỗi JSON (không pickle) để nạp an toàn."""
    arrays = {
        "order": np.array(lm.order),
        "symbols_json": np.array(json.dumps(lm.alphabet.symbols, ensure_ascii=False)),
    }
    for k, t in enumerate(lm.tables, start=1):
        arrays[f"h{k}"], arrays[f"p{k}"], arrays[f"b{k}"] = t.hashes, t.logp, t.backoff
    np.savez(path, **arrays)


def load_bin(path: Path) -> CharNGramLM:
    z = np.load(path)  # không allow_pickle: chỉ mảng số và một chuỗi unicode
    order = int(z["order"])
    alphabet = LMAlphabet(json.loads(str(z["symbols_json"])))
    tables = [NGramTable(z[f"h{k}"], z[f"p{k}"], z[f"b{k}"]) for k in range(1, order + 1)]
    return CharNGramLM(alphabet, order, tables)
```

```python
"""Chuẩn hóa corpus theo profile: viết hoa, alias, chỉ giữ bảng chữ LM, một câu một dòng.

Xuất ba file: <lang>.txt (thuần), <lang>.chars.txt (token hóa ký tự cho lmplz, khoảng trắng = ▁),
<lang>.holdout.txt (mỗi dòng thứ holdout_every, không vào train, để đo perplexity).
"""

from __future__ import annotations

import argparse
import json
import re
from collections import Counter
from pathlib import Path

from morsedesk.encode.text import normalize
from morsedesk.lm.alphabet import LMAlphabet
from morsedesk.lm.profile import Profile, load_profile
from morsedesk.textio import force_utf8_streams

TYPO_MAP = {
    "’": "'", "‘": "'", "“": '"', "”": '"', "«": '"', "»": '"',
    "„": '"', "—": "-", "–": "-", "…": "...", " ": " ",
}
_SENT_RE = re.compile(r"(?<=[.!?])\s+")


def split_sentences(text: str) -> list[str]:
    return [s.strip() for s in _SENT_RE.split(text) if s.strip()]


def normalize_line(line: str, profile: Profile, alphabet: LMAlphabet,
                   max_unknown_ratio: float = 0.05, min_len: int = 3) -> str | None:
    for src, dst in TYPO_MAP.items():
        line = line.replace(src, dst)
    text = normalize(line, profile)  # viết hoa + alias + gộp khoảng trắng
    if not text:
        return None
    non_space = [ch for ch in text if ch != " "]
    unknown = sum(1 for ch in non_space if ch not in alphabet.index)
    if not non_space or unknown / len(non_space) > max_unknown_ratio:
        return None
    kept = "".join(ch for ch in text if ch == " " or ch in alphabet.index)
    kept = " ".join(kept.split())
    return kept if len(kept) >= min_len else None


def normalize_paths(paths: list[Path], profile_id: str, out_plain: Path, out_chars: Path,
                    out_holdout: Path, holdout_every: int = 500, max_lines: int | None = None) -> dict:
    profile = load_profile(profile_id)
    alphabet = LMAlphabet.for_profile(profile)
    stats = {"lines_in": 0, "lines_out": 0, "holdout": 0, "chars": 0}
    hist: Counter = Counter()
    for p in (out_plain, out_chars, out_holdout):
        p.parent.mkdir(parents=True, exist_ok=True)
    kept_index = 0
    with open(out_plain, "w", encoding="utf-8") as fp, open(out_chars, "w", encoding="utf-8") as fc, \
            open(out_holdout, "w", encoding="utf-8") as fh:
        for path in paths:
            for raw in Path(path).read_text(encoding="utf-8", errors="replace").splitlines():
                for sent in split_sentences(raw) or [raw]:
                    if not sent.strip():
                        continue
                    stats["lines_in"] += 1
                    norm = normalize_line(sent, profile, alphabet)
                    if norm is None:
                        continue
                    kept_index += 1
                    if kept_index % holdout_every == 0:
                        fh.write(norm + "\n")
                        stats["holdout"] += 1
                        continue
                    fp.write(norm + "\n")
                    fc.write(alphabet.to_tokens_line(norm) + "\n")
                    stats["lines_out"] += 1
                    stats["chars"] += len(norm)
                    hist.update(norm)
                    if max_lines and stats["lines_out"] >= max_lines:
                        break
    stats["top_chars"] = hist.most_common(20)
    return stats


def main(argv: list[str] | None = None) -> int:
    force_utf8_streams()
    ap = argparse.ArgumentParser(description="Chuẩn hóa corpus theo profile")
    ap.add_argument("--profile", required=True)
    ap.add_argument("--lang", required=True, help="tên ngắn cho file ra: en, ru, kk")
    ap.add_argument("--in", dest="inputs", nargs="+", type=Path, required=True, help="file hoặc thư mục")
    ap.add_argument("--out-dir", type=Path, default=Path("corpus"))
    ap.add_argument("--holdout-every", type=int, default=500)
    ap.add_argument("--max-lines", type=int, default=None)
    a = ap.parse_args(argv)
    paths: list[Path] = []
    for p in a.inputs:
        paths.extend(sorted(p.glob("*.txt")) if p.is_dir() else [p])
    stats = normalize_paths(paths, a.profile, a.out_dir / f"{a.lang}.txt", a.out_dir / f"{a.lang}.chars.txt",
                            a.out_dir / f"{a.lang}.holdout.txt", a.holdout_every, a.max_lines)
    print(json.dumps(stats, ensure_ascii=False))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
"""Tải corpus LM từ nguồn giấy phép rộng (public domain, CC BY, MIT, Apache-2.0).

Ghi corpus/raw/<lang>/*.txt và corpus/raw/<lang>/SOURCES.json (tên, URL, license, số dòng).
Chỉ chạy trên máy dev có mạng; không dùng trong runtime.
"""

from __future__ import annotations

import argparse
import io
import json
import re
import time
import urllib.request
from pathlib import Path

import numpy as np

from morsedesk.data.text import TextSampler
from morsedesk.lm.profile import load_profile
from morsedesk.textio import force_utf8_streams

UA = {"User-Agent": "morsedesk-corpus-fetch/0.1 (+offline morse decoder research)"}
LANG_PROFILE = {"en": "en", "ru": "ru", "kk": "kk-cyrl-ru"}
GUTENBERG_TOP = "https://www.gutenberg.org/browse/scores/top"
GUTENBERG_RU = "https://www.gutenberg.org/browse/languages/ru"
GUTENBERG_TXT = "https://www.gutenberg.org/cache/epub/{id}/pg{id}.txt"
HF_PARQUET_API = "https://huggingface.co/api/datasets/{repo}/parquet/{config}/{split}"
HF_TREE_API = "https://huggingface.co/api/datasets/{repo}/tree/main"
HF_RESOLVE = "https://huggingface.co/datasets/{repo}/resolve/main/{path}"
_ID_RE = re.compile(r'href="/ebooks/(\d+)"')
_START_RE = re.compile(r"\*\*\* ?START OF[^\n]*\*\*\*")
_END_RE = re.compile(r"\*\*\* ?END OF[^\n]*\*\*\*")


def _get(url: str, timeout: float = 60.0) -> bytes:
    req = urllib.request.Request(url, headers=UA)
    with urllib.request.urlopen(req, timeout=timeout) as r:
        return r.read()


def gutenberg_ids_from_html(html: str) -> list[int]:
    seen: list[int] = []
    for m in _ID_RE.finditer(html):
        i = int(m.group(1))
        if i not in seen:
            seen.append(i)
    return seen


def strip_gutenberg(text: str) -> str:
    s = _START_RE.search(text)
    e = _END_RE.search(text)
    if s and e and e.start() > s.end():
        return text[s.end():e.start()]
    return text


def detokenize(tokens: list[str]) -> str:
    out = ""
    for tok in tokens:
        if tok in {",", ".", "!", "?", ":", ";", ")"} or not out:
            out += tok
        elif out.endswith("("):
            out += tok
        else:
            out += " " + tok
    return out


def fetch_gutenberg(ids: list[int], out_dir: Path, sleep_s: float, sources: list) -> int:
    out_dir.mkdir(parents=True, exist_ok=True)
    n = 0
    for i in ids:
        dest = out_dir / f"gutenberg_{i}.txt"
        if dest.exists():
            n += 1
            continue
        try:
            text = _get(GUTENBERG_TXT.format(id=i)).decode("utf-8", errors="replace")
        except Exception as e:  # nguồn lẻ lỗi không chặn cả lô
            print(f"bỏ qua Gutenberg {i}: {e}")
            continue
        body = strip_gutenberg(text)
        dest.write_text(body, encoding="utf-8")
        sources.append({"name": f"Project Gutenberg #{i}", "url": GUTENBERG_TXT.format(id=i),
                        "license": "Public domain (Project Gutenberg License)", "lines": body.count("\n")})
        n += 1
        time.sleep(sleep_s)
    return n


def fetch_parquet_urls(repo: str, config: str = "default", split: str = "train") -> list[str]:
    return json.loads(_get(HF_PARQUET_API.format(repo=repo, config=config, split=split)))


def parquet_column_lines(url: str, column: str, join_tokens: bool = False) -> list[str]:
    import pyarrow.parquet as pq

    table = pq.read_table(io.BytesIO(_get(url, timeout=600.0)), columns=[column])
    values = table.column(column).to_pylist()
    if join_tokens:
        return [detokenize([str(t) for t in v]) for v in values if v]
    return [str(v) for v in values if v]


def write_ham_lines(profile_id: str, n: int, out: Path, seed: int = 0) -> int:
    """Radiogram tự sinh (HAM_TEMPLATES, callsign ngẫu nhiên) đã chuyển tự sang profile."""
    profiles = {pid: load_profile(pid) for pid in {"en", profile_id}}
    sampler = TextSampler(profiles, ham_ratio=1.0, random_ratio=0.0)
    rng = np.random.default_rng(seed)
    lines = [sampler.sample(profile_id, rng) for _ in range(n)]
    return _write_lines(out, lines)


def _write_lines(path: Path, lines: list[str]) -> int:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text("\n".join(lines) + "\n", encoding="utf-8")
    return len(lines)


def fetch_lang(lang: str, out_dir: Path, limit_books: int, sleep_s: float, ham_lines: int = 20000) -> None:
    dest = out_dir / lang
    dest.mkdir(parents=True, exist_ok=True)
    sources: list[dict] = []
    if lang == "en":
        ids = gutenberg_ids_from_html(_get(GUTENBERG_TOP).decode("utf-8", errors="replace"))[:limit_books]
```

```python
"""Nạp và cache LM n-gram theo profile từ thư mục models/lm."""

from __future__ import annotations

from pathlib import Path

from morsedesk.lm.ngram import CharNGramLM, load_bin
from morsedesk.lm.profile import Profile, load_profile


def default_lm_dir() -> Path:
    return Path(__file__).resolve().parents[2] / "models" / "lm"


def lm_file_for(profile: Profile, lm_dir: Path) -> Path:
    return Path(lm_dir) / Path(profile.lm).name


class LMStore:
    def __init__(self, lm_dir: Path | None = None) -> None:
        self.lm_dir = Path(lm_dir) if lm_dir else default_lm_dir()
        self._lms: dict[str, CharNGramLM] = {}
        self._profiles: dict[str, Profile] = {}

    def get_profile(self, profile_id: str) -> Profile:
        if profile_id not in self._profiles:
            self._profiles[profile_id] = load_profile(profile_id)
        return self._profiles[profile_id]

    def register(self, profile: Profile) -> None:
        self._profiles[profile.id] = profile

    def available(self, profile_id: str) -> bool:
        return lm_file_for(self.get_profile(profile_id), self.lm_dir).exists()

    def get(self, profile_id: str) -> CharNGramLM:
        if profile_id not in self._lms:
            path = lm_file_for(self.get_profile(profile_id), self.lm_dir)
            if not path.exists():
                raise FileNotFoundError(f"thiếu mô hình ngôn ngữ {path}; chạy tools/build_lm.sh và tools/arpa_to_bin.py")
            self._lms[profile_id] = load_bin(path)
        return self._lms[profile_id]
```

```python
"""ARPA (từ lmplz) -> nhị phân numpy .bin theo bảng chữ của profile, kèm thống kê và perplexity."""

from __future__ import annotations

import argparse
import json
from pathlib import Path

from morsedesk.lm.alphabet import LMAlphabet
from morsedesk.lm.ngram import load_arpa, save_bin
from morsedesk.lm.profile import load_profile
from morsedesk.textio import force_utf8_streams


def convert(arpa: Path, profile_id: str, out: Path, holdout: Path | None = None, holdout_lines: int = 1000) -> dict:
    alphabet = LMAlphabet.for_profile(load_profile(profile_id))
    lm = load_arpa(arpa, alphabet)
    out.parent.mkdir(parents=True, exist_ok=True)
    save_bin(lm, out)
    stats = {
        "arpa": str(arpa), "out": str(out), "order": lm.order,
        "ngrams_per_order": [int(len(t.hashes)) for t in lm.tables],
        "size_mb": round(out.stat().st_size / 1e6, 2),
        "perplexity": None,
    }
    if holdout and Path(holdout).exists():
        lines = Path(holdout).read_text(encoding="utf-8").splitlines()[:holdout_lines]
        stats["perplexity"] = round(lm.perplexity([ln for ln in lines if ln.strip()]), 3)
    return stats


def main(argv: list[str] | None = None) -> int:
    force_utf8_streams()
    ap = argparse.ArgumentParser(description="ARPA -> .bin")
    ap.add_argument("--arpa", type=Path, required=True)
    ap.add_argument("--profile", required=True)
    ap.add_argument("--out", type=Path, required=True)
    ap.add_argument("--holdout", type=Path, default=None)
    a = ap.parse_args(argv)
    print(json.dumps(convert(a.arpa, a.profile, a.out, a.holdout), ensure_ascii=False, indent=2))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
"""Dựng LM nhỏ thuần Python từ corpus đã chuẩn hóa (dự phòng khi không có KenLM). Cắt còn max_mb đầu."""

from __future__ import annotations

import argparse
import json
from pathlib import Path

from morsedesk.lm.alphabet import LMAlphabet
from morsedesk.lm.ngram import save_bin, train_small
from morsedesk.lm.profile import load_profile
from morsedesk.textio import force_utf8_streams


def build(text: Path, profile_id: str, out: Path, order: int = 5, max_mb: float = 5.0) -> dict:
    lines: list[str] = []
    used = 0
    limit = int(max_mb * 1e6)
    with open(text, encoding="utf-8") as f:
        for ln in f:
            ln = ln.rstrip("
")
            used += len(ln.encode("utf-8")) + 1
            if used > limit:
                break
            if ln.strip():
                lines.append(ln)
    lm = train_small(lines, LMAlphabet.for_profile(load_profile(profile_id)), order=order)
    out.parent.mkdir(parents=True, exist_ok=True)
    save_bin(lm, out)
    return {"lines": len(lines), "order": order, "size_mb": round(out.stat().st_size / 1e6, 2), "out": str(out)}


def main(argv: list[str] | None = None) -> int:
    force_utf8_streams()
    ap = argparse.ArgumentParser(description="LM nhỏ thuần Python")
    ap.add_argument("--profile", required=True)
    ap.add_argument("--text", type=Path, required=True)
    ap.add_argument("--out", type=Path, required=True)
    ap.add_argument("--order", type=int, default=5)
    ap.add_argument("--max-mb", type=float, default=5.0)
    a = ap.parse_args(argv)
    print(json.dumps(build(a.text, a.profile, a.out, a.order, a.max_mb), ensure_ascii=False))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
from morsedesk.model.vocab import VOCAB  # noqa: E402


def lattice_from_tokens(tokens: list[str], gap: int = 6, peak: float = 0.9, n_classes: int = len(VOCAB),
                        lead: int = 3) -> np.ndarray:
    """Lưới CTC nhọn: `lead` khung blank, rồi mỗi token một khung đỉnh + `gap` khung blank."""
    rest = np.log((1.0 - peak) / (n_classes - 1))
    rows = []

    def one_hot(c: int) -> np.ndarray:
        r = np.full(n_classes, rest, dtype=np.float32)
        r[c] = np.log(peak)
        return r

    rows += [one_hot(VOCAB.blank_id)] * lead
    for tok in tokens:
        rows.append(one_hot(VOCAB.id(tok)))
        rows += [one_hot(VOCAB.blank_id)] * gap
    return np.stack(rows)
```

```python
"""Prefix beam search CTC trên lưới 65 lớp mẫu; rẽ nhánh chữ theo profile, chấm bằng LM ký tự.

Tiền tố = (chuỗi token đã phát, chuỗi chữ đã chọn). Mỗi khung chỉ xét blank cộng tối đa
`topk_per_frame` lớp có xác suất >= `min_prob`. Khung không có lớp nào đạt ngưỡng đi đường tắt
chỉ-blank (không sinh tiền tố mới), nhờ đó chi phí tỉ lệ với số token chứ không phải số khung.
"""

from __future__ import annotations

import math
from dataclasses import dataclass

import numpy as np

from morsedesk.lm.alphabet import LMAlphabet
from morsedesk.lm.ngram import CharNGramLM
from morsedesk.lm.profile import UNK_DISPLAY, Profile
from morsedesk.model.vocab import SPACE, UNK, VOCAB

NEG_INF = -math.inf


def _lse(a: float, b: float) -> float:
    if a == NEG_INF:
        return b
    if b == NEG_INF:
        return a
    m = a if a > b else b
    return m + math.log(math.exp(a - m) + math.exp(b - m))


@dataclass(frozen=True)
class BeamParams:
    beam: int = 64
    alpha: float = 0.6
    beta: float = 0.5
    lexicon_bonus: float = 1.0
    topk_per_frame: int = 8
    unk_logp: float = math.log(1e-4)
    min_prob: float = 1e-3
    space_id: int = VOCAB.space_id
    blank_id: int = VOCAB.blank_id


@dataclass
class Hypothesis:
    tokens: tuple[str, ...]
    letters: tuple[str, ...]
    frames: tuple[int, ...]
    probs: tuple[float, ...]
    acoustic_logp: float
    lm_logp: float
    score: float
    n_words: int

    @property
    def text(self) -> str:
        return "".join(self.letters)

    @property
    def n_chars(self) -> int:
        return len(self.tokens)

    @property
    def emissions(self) -> list[tuple[str, int, float]]:
        return list(zip(self.tokens, self.frames, self.probs, strict=True))


def class_options(profile: Profile, alphabet: LMAlphabet) -> list[list[tuple[str, int | None]]]:
    """Với mỗi id lớp: danh sách (chuỗi hiển thị, id LM). id LM = None cho `#` và prosign."""
    out: list[list[tuple[str, int | None]]] = []
    for c, tok in enumerate(VOCAB.tokens):
        if c == VOCAB.blank_id:
            out.append([])
        elif c == VOCAB.space_id:
            out.append([(" ", alphabet.space_id)])
        elif c == VOCAB.unk_id:
            out.append([(UNK_DISPLAY, None)])
        elif tok in profile.map:
            out.append([(ch, alphabet.index[ch]) for ch in profile.decode_options(tok)])
        else:
            opts = profile.decode_options(tok)
            out.append([(UNK_DISPLAY if opts[0] == UNK else opts[0], None)])
    return out


class _Prefix:
    __slots__ = ("tokens", "letters", "frames", "probs", "p_b", "p_nb",
                 "lm_ctx", "lm_logp", "bonus", "n_words", "word")

    def __init__(self) -> None:
        self.tokens: tuple[str, ...] = ()
        self.letters: tuple[str, ...] = ()
        self.frames: tuple[int, ...] = ()
        self.probs: tuple[float, ...] = ()
        self.p_b = NEG_INF
        self.p_nb = NEG_INF
        self.lm_ctx: tuple[int, ...] = ()
        self.lm_logp = 0.0
        self.bonus = 0.0
        self.n_words = 0
        self.word = ""

    def total(self, alpha: float) -> float:
        return _lse(self.p_b, self.p_nb) + alpha * self.lm_logp + self.bonus

    def clone_reset(self) -> _Prefix:
        c = _Prefix()
        c.tokens, c.letters, c.frames, c.probs = self.tokens, self.letters, self.frames, self.probs
        c.lm_ctx, c.lm_logp, c.bonus, c.n_words, c.word = (
            self.lm_ctx, self.lm_logp, self.bonus, self.n_words, self.word)
        return c


class BeamState:
    def __init__(self, profile: Profile, lm: CharNGramLM | None = None, params: BeamParams | None = None) -> None:
        self.profile = profile
        self.lm = lm
        self.params = params or BeamParams()
        expected = LMAlphabet.for_profile(profile)
        if lm is not None and lm.alphabet.symbols != expected.symbols:
            raise ValueError(f"bảng chữ của LM không khớp profile {profile.id}")
        self.alphabet = lm.alphabet if lm is not None else expected
        self.options = class_options(profile, self.alphabet)
        self.lexicon = set(profile.lexicon_bonus)
        self.ctx_len = max(0, lm.order - 1) if lm is not None else 0
        self.t = 0
        self._lm_cache: dict[tuple[tuple[int, ...], tuple[int, ...]], dict[int, float]] = {}
        root = _Prefix()
        root.p_b = 0.0
        root.lm_ctx = (self.alphabet.bos_id,)
        self.beams: dict[tuple[tuple[str, ...], tuple[str, ...]], _Prefix] = {((), ()): root}

    # ---- LM ----
    def _lm_scores(self, pre: _Prefix, classes: list[int]) -> dict[int, float]:
        if self.lm is None:
            return {}
        ids = sorted({lm_id for c in classes for _, lm_id in self.options[c] if lm_id is not None})
        if not ids:
            return {}
        key = (pre.lm_ctx, tuple(ids))
        hit = self._lm_cache.get(key)
        if hit is None:
            vals = self.lm.score_many(pre.lm_ctx, np.asarray(ids, dtype=np.int64))
            hit = {i: float(v) for i, v in zip(ids, vals, strict=True)}
            if len(self._lm_cache) > 50_000:
                self._lm_cache.clear()
            self._lm_cache[key] = hit
        return hit

    def _extend(self, pre: _Prefix, tok: str, display: str, lm_id: int | None,
                lm_scores: dict[int, float], frame: int, prob: float) -> _Prefix:
        p = self.params
        ext = _Prefix()
        ext.tokens = pre.tokens + (tok,)
        ext.letters = pre.letters + (display,)
        ext.frames = pre.frames + (frame,)
        ext.probs = pre.probs + (prob,)
        ext.lm_ctx, ext.lm_logp, ext.bonus, ext.n_words, ext.word = (
            pre.lm_ctx, pre.lm_logp, pre.bonus, pre.n_words, pre.word)
        if lm_id is None:                                   # `#` hoặc prosign: phạt, không đổi context
            if self.lm is not None:
                ext.lm_logp += p.unk_logp
            return ext
        if self.lm is not None:
            ext.lm_logp += lm_scores[lm_id]
            ext.lm_ctx = (pre.lm_ctx + (lm_id,))[-self.ctx_len:] if self.ctx_len else ()
        if tok == SPACE:
            if pre.tokens and pre.tokens[-1] != SPACE:      # kết thúc một từ
                ext.n_words += 1
                ext.bonus += p.beta
                if pre.word in self.lexicon:
                    ext.bonus += p.lexicon_bonus
            ext.word = ""
        else:
            ext.word = pre.word + display
        return ext

    # ---- một khung ----
    def _advance(self, lp: np.ndarray) -> None:
        p = self.params
        lp_blank = float(lp[p.blank_id])
        k = min(p.topk_per_frame + 1, len(lp))
        cand = np.argpartition(-lp, k - 1)[:k]
        thresh = math.log(p.min_prob)
        classes = [int(c) for c in cand if int(c) != p.blank_id and lp[c] >= thresh]
        if not classes:                                     # đường tắt chỉ-blank
            for pre in self.beams.values():
                pre.p_b = _lse(pre.p_b, pre.p_nb) + lp_blank
                pre.p_nb = NEG_INF
            return
        classes.sort(key=lambda c: -float(lp[c]))
        classes = classes[: p.topk_per_frame]
        nxt: dict[tuple[tuple[str, ...], tuple[str, ...]], _Prefix] = {}
        for key, pre in self.beams.items():
            same = nxt.get(key)
            if same is None:
                same = pre.clone_reset()
                nxt[key] = same
            same.p_b = _lse(same.p_b, _lse(pre.p_b, pre.p_nb) + lp_blank)
            last = pre.tokens[-1] if pre.tokens else None
            lm_scores = self._lm_scores(pre, classes)
            for c in classes:
                tok = VOCAB.token(c)
                lp_c = float(lp[c])
                if tok == last:
                    same.p_nb = _lse(same.p_nb, pre.p_nb + lp_c)
                    p_new = pre.p_b + lp_c
                else:
                    p_new = _lse(pre.p_b, pre.p_nb) + lp_c
                if p_new == NEG_INF:
                    continue
                for display, lm_id in self.options[c]:
                    new_key = (pre.tokens + (tok,), pre.letters + (display,))
                    ext = nxt.get(new_key)
                    if ext is None:
                        ext = self._extend(pre, tok, display, lm_id, lm_scores, self.t, math.exp(lp_c))
                        nxt[new_key] = ext
                    ext.p_nb = _lse(ext.p_nb, p_new)
        ranked = sorted(nxt.values(), key=lambda h: h.total(p.alpha), reverse=True)[: p.beam]
        self.beams = {(h.tokens, h.letters): h for h in ranked}

    def step(self, log_probs: np.ndarray) -> BeamState:
        for row in np.asarray(log_probs, dtype=np.float32):
            self._advance(row)
            self.t += 1
        return self

    def hypotheses(self) -> list[Hypothesis]:
        p = self.params
        out: list[Hypothesis] = []
        for pre in self.beams.values():
            acoustic = _lse(pre.p_b, pre.p_nb)
            lm_logp = pre.lm_logp
            bonus = pre.bonus
            n_words = pre.n_words
            if self.lm is not None:
                lm_logp += float(self.lm.score(pre.lm_ctx, self.alphabet.eos_id))
            if pre.word:                                    # từ cuối chưa đóng bằng khoảng trắng
                n_words += 1
                bonus += p.beta + (p.lexicon_bonus if pre.word in self.lexicon else 0.0)
            out.append(Hypothesis(pre.tokens, pre.letters, pre.frames, pre.probs, acoustic, lm_logp,
                                  acoustic + p.alpha * lm_logp + bonus, n_words))
        out.sort(key=lambda h: h.score, reverse=True)
        return out


def beam_search(log_probs: np.ndarray, profile: Profile, lm: CharNGramLM | None = None,
                params: BeamParams | None = None) -> list[Hypothesis]:
    return BeamState(profile, lm, params).step(log_probs).hypotheses()
```

```python
"""Nhận diện ngôn ngữ theo đoạn: so điểm LM chuẩn hóa của giả thuyết beam tốt nhất giữa các profile."""

from __future__ import annotations

from collections import defaultdict
from dataclasses import dataclass, field

import numpy as np

from morsedesk.lm.beam import BeamParams, Hypothesis, beam_search
from morsedesk.lm.ngram import CharNGramLM
from morsedesk.lm.profile import Profile
from morsedesk.model.vocab import VOCAB

SILENCE_FRAMES = 300   # 3 s ở 100 khung đầu ra/s
MAX_FRAMES = 3000      # 30 s
MIN_MARGIN = 0.15      # nat/ký tự


@dataclass
class Segment:
    start: int
    end: int
    profile_id: str | None
    scores: dict[str, float]
    n_chars: int
    best: dict[str, Hypothesis] = field(default_factory=dict)


@dataclass
class LangIdResult:
    profile_id: str
    scores: dict[str, float]
    segments: list[Segment]


def split_segments(log_probs: np.ndarray, silence_frames: int = SILENCE_FRAMES,
                   max_frames: int = MAX_FRAMES, blank_id: int = VOCAB.blank_id) -> list[tuple[int, int]]:
    n = int(log_probs.shape[0])
    if n == 0:
        return []
    is_blank = log_probs.argmax(axis=1) == blank_id
    cuts: list[int] = []
    run_start = None
    for t in range(n + 1):
        blank = t < n and bool(is_blank[t])
        if blank and run_start is None:
            run_start = t
        elif not blank and run_start is not None:
            if t - run_start >= silence_frames and run_start > 0 and t < n:
                cuts.append((run_start + t) // 2)
            run_start = None
    bounds = [0, *cuts, n]
    segments: list[tuple[int, int]] = []
    for s, e in zip(bounds[:-1], bounds[1:], strict=True):
        parts = max(1, -(-(e - s) // max_frames))
        edges = np.linspace(s, e, parts + 1).astype(int)
        segments += [(int(a), int(b)) for a, b in zip(edges[:-1], edges[1:], strict=True) if b > a]
    return segments


def detect(log_probs: np.ndarray, profiles: dict[str, Profile], lms: dict[str, CharNGramLM],
           params: BeamParams | None = None, min_margin: float = MIN_MARGIN,
           silence_frames: int = SILENCE_FRAMES, max_frames: int = MAX_FRAMES) -> LangIdResult:
    params = params or BeamParams()
    segments: list[Segment] = []
    prev: str | None = None
    for s, e in split_segments(log_probs, silence_frames, max_frames, params.blank_id):
        best = {pid: beam_search(log_probs[s:e], prof, lms[pid], params)[0] for pid, prof in profiles.items()}
        scores = {pid: h.lm_logp / h.n_chars for pid, h in best.items() if h.n_chars > 0}
        if not scores:
            segments.append(Segment(s, e, None, {}, 0, best))
            continue
        ranked = sorted(scores.items(), key=lambda kv: kv[1], reverse=True)
        chosen = ranked[0][0]
        margin = ranked[0][1] - ranked[1][1] if len(ranked) > 1 else float("inf")
        if margin < min_margin and prev is not None:
            chosen = prev
        segments.append(Segment(s, e, chosen, scores, best[chosen].n_chars, best))
        prev = chosen
    votes: dict[str, int] = defaultdict(int)
    weighted: dict[str, float] = defaultdict(float)
    weight_total = 0
    for seg in segments:
        if seg.profile_id is None:
            continue
        votes[seg.profile_id] += seg.n_chars
        weight_total += seg.n_chars
        for pid, sc in seg.scores.items():
            weighted[pid] += sc * seg.n_chars
    if not votes:
        return LangIdResult(next(iter(profiles)), {}, segments)
    file_pid = max(votes.items(), key=lambda kv: kv[1])[0]
    scores = {pid: weighted[pid] / weight_total for pid in profiles if pid in weighted}
    return LangIdResult(file_pid, scores, segments)
```

```python
        i += 1
    return " ".join("".join(out).split())
```

```python
                       target_s: float) -> tuple[list[str], str]:
        """(docstring giữ nguyên) ... Trả (tokens, text_original) với text_original là các từ gốc đã ghép."""
        tokens: list[str] = []
        words_text: list[str] = []
        queue: list[tuple[list[str], str]] = []
        rounds = 0
        while rounds < MAX_SAMPLE_ROUNDS:
            if not queue:
                rounds += 1
                text = self.text.sample(profile.id, rng)
                queue = [(text_to_tokens(w, profile), w) for w in text.split()]
                queue = [(t, w) for t, w in queue if t]
                continue
            word, word_text = queue.pop(0)
            cand = tokens + ([SPACE] if tokens else []) + word
            d = self.tokens_duration(cand, keyer)
            if d <= self.cfg.max_seconds:
                tokens = cand
                words_text.append(word_text)
                if d >= target_s:
                    break
            elif self.tokens_duration(tokens, keyer) >= self.cfg.min_seconds:
                break
        if not tokens:
            tokens, word_text = self._shortest_builtin_word(profile, keyer)
            words_text = [word_text]
        return tokens, " ".join(words_text)
```

```python
def _group_words(items: list[tuple[str, str, int, float]], frame_rate: int, end_frame: int) -> list[Word]:
    """items: (token, chữ hiển thị, khung đầu vào, xác suất). Tách từ tại SPACE; tin cậy = trung bình nhân."""
    words: list[Word] = []
    cur: list[str] = []
    probs: list[float] = []
    start = 0
    for tok, letter, f, p in list(items) + [(SPACE, " ", end_frame, 1.0)]:
        if tok == SPACE:
            if cur:
                conf = float(np.exp(np.mean(np.log(np.maximum(probs, 1e-6)))))
                words.append(Word("".join(cur), start / frame_rate, f / frame_rate, conf))
                cur, probs = [], []
            continue
        if not cur:
            start = f
        cur.append(letter)
        probs.append(p)
    return words


def emissions_to_words(
    emissions: list[tuple[str, int, float]], profile: Profile, frame_rate: int, end_frame: int
) -> list[Word]:
    return _group_words([(tok, profile.render([tok]), f, p) for tok, f, p in emissions], frame_rate, end_frame)


def hypothesis_to_words(hyp: Hypothesis, frame_rate: int, stride: int, end_frame: int, offset: int = 0) -> list[Word]:
    """Từ từ giả thuyết beam: chữ đã chọn theo LM; khung token = (offset + khung ra) * stride."""
    items = [(tok, letter, (offset + f) * stride, p)
             for tok, letter, f, p in zip(hyp.tokens, hyp.letters, hyp.frames, hyp.probs, strict=True)]
    return _group_words(items, frame_rate, end_frame)


def _strip_spaces(emissions: list[tuple[str, int, float]]) -> list[tuple[str, int, float]]:
    i, j = 0, len(emissions)
    while i < j and emissions[i][0] == SPACE:
        i += 1
    while j > i and emissions[j - 1][0] == SPACE:
        j -= 1
    return emissions[i:j]
```

```python
class Decoder:
    def __init__(
        self, profile: str | Profile = "en", engine: str = "baseline", model_path: Path | None = None,
        fallback: bool = True, lm: bool = True, beam: int = 64, lm_dir: Path | None = None,
        acoustic_model=None,
    ) -> None:
        self.auto = isinstance(profile, str) and profile == "auto"
        self.profile: Profile | None = None if self.auto else _resolve_profile(profile)
        self.engine = engine
        self.fallback = fallback
        self.lm_enabled = bool(lm)
        self.beam_params = BeamParams(beam=int(beam))
        self._baseline = BaselineDecoder()
        self._model = acoustic_model
        if engine == "onnx":
            if self._model is None:
                from morsedesk.model.onnx_runner import OnnxAcousticModel

                self.model_path = Path(model_path) if model_path else DEFAULT_MODEL_PATH
                self._model = OnnxAcousticModel(self.model_path)
        elif engine != "baseline":
            raise NotImplementedError(f"engine chưa hỗ trợ: {engine}")
        if self.auto and not self.lm_enabled:
            raise ValueError("profile 'auto' cần mô hình ngôn ngữ; bỏ --no-lm hoặc chọn profile cụ thể")
        self._store = LMStore(lm_dir)
        if self.profile is not None:
            self._store.register(self.profile)
        self.candidates: list[str] = list_profiles() if self.auto else [self.profile.id]
        self._profiles: dict[str, Profile] = {pid: self._store.get_profile(pid) for pid in self.candidates}
        self._lms: dict[str, CharNGramLM] = {}
        if self.lm_enabled and (self.auto or engine == "onnx"):
            for pid in self.candidates:
                self._lms[pid] = self._store.get(pid)   # thiếu file -> FileNotFoundError nêu đường dẫn

    def _empty(self, warnings: list[str]) -> DecodeResult:
        lang = self.profile.id if self.profile is not None else ""
        return DecodeResult("", [], None, None, None, lang, {}, warnings, self.engine)

    def decode_array(self, x: np.ndarray, sample_rate: int = SAMPLE_RATE) -> DecodeResult:
        if sample_rate != SAMPLE_RATE:
            x = resample(np.asarray(x, dtype=np.float32), sample_rate, SAMPLE_RATE)
        est = estimate_tone(x)
        if not est.has_signal or est.f0_hz is None:
            return self._empty(["không có tín hiệu Morse trong đoạn âm thanh"])
        warnings: list[str] = []
        if est.second_peak_hz is not None:
            warnings.append(
                f"phát hiện tone thứ hai tại {est.second_peak_hz:.0f} Hz, đang bám {est.f0_hz:.0f} Hz"
            )
        if self.engine == "baseline":
            return self._decode_baseline(x, est, warnings)
        return self._decode_onnx(x, est, warnings)

    def _baseline_tokens(self, x: np.ndarray, f0_hz: float):
        return self._baseline.decode_envelope(mix_envelope(x, f0_hz))

    def _pick_profile_for_tokens(self, tokens: list[str]) -> tuple[Profile, dict[str, float]]:
        """Chế độ auto với engine baseline: chấm văn bản render qua từng profile bằng LM."""
        if not self.auto:
            return self.profile, {}
        if not tokens:
            return self._profiles[self.candidates[0]], {}
        scores = {}
        for pid in self.candidates:
            text = self._profiles[pid].render(tokens)
            scores[pid] = self._lms[pid].sentence_logprob(text) / (len(text) + 1)
        best = max(scores, key=scores.get)
        return self._profiles[best], scores

    def _decode_baseline(self, x, est, warnings: list[str]) -> DecodeResult:
        res = self._baseline_tokens(x, est.f0_hz)
        profile, scores = self._pick_profile_for_tokens(res.tokens)
        confidence = float(np.exp(-res.cost)) if np.isfinite(res.cost) else 0.0
        if confidence < LOW_CONFIDENCE:
            warnings.append("độ tin cậy thấp")
        words = symbols_to_words(res.symbols, profile, FRAME_RATE, confidence)
        return DecodeResult(" ".join(w.text for w in words), words, res.wpm, est.f0_hz, est.snr_db,
                            profile.id, scores, warnings, "baseline")

    def _log_probs(self, x: np.ndarray, f0_hz: float) -> tuple[np.ndarray, int]:
        # Nếu kế hoạch tuyến B (Task 1) đã tạo hàm này với nhánh feature_mode == "window",
        # GIỮ NGUYÊN thân hàm đó; bản dưới là bản chỉ có đường theo clip.
        from morsedesk.model.chunker import run_chunked

        feats = compute_features(x, f0_hz)
        return run_chunked(self._model, feats), len(feats)

    def _beam_words(self, logp: np.ndarray, stride: int, n_in: int):
        if not self.auto:
            hyp = beam_search(logp, self.profile, self._lms[self.profile.id], self.beam_params)[0]
            emissions = _strip_spaces([(tok, f * stride, p) for tok, f, p in hyp.emissions])
            return hypothesis_to_words(hyp, FRAME_RATE, stride, n_in), emissions, self.profile, {}
        res = detect(logp, self._profiles, self._lms, self.beam_params)
        words: list[Word] = []
        emissions: list[tuple[str, int, float]] = []
        for seg in res.segments:
            if seg.profile_id is None:
                continue
            hyp = seg.best[seg.profile_id]
            end_in = min(n_in, seg.end * stride)
            words += hypothesis_to_words(hyp, FRAME_RATE, stride, end_in, offset=seg.start)
            emissions += [(tok, (seg.start + f) * stride, p) for tok, f, p in hyp.emissions]
        return words, _strip_spaces(emissions), self._profiles[res.profile_id], res.scores

    def _decode_onnx(self, x, est, warnings: list[str]) -> DecodeResult:
        logp, n_in = self._log_probs(x, est.f0_hz)
        stride = int(self._model.time_stride)
        if self.lm_enabled:
            words, emissions, profile, scores = self._beam_words(logp, stride, n_in)
        else:
            profile, scores = self.profile, {}
            emissions = _strip_spaces([(VOCAB.token(i), f * stride, p)
                                       for i, f, p in greedy_decode_with_frames(logp, VOCAB.blank_id)])
            words = emissions_to_words(emissions, profile, FRAME_RATE, n_in)
        confidence = float(np.mean([w.confidence for w in words])) if words else 0.0
        fallback_text = None
        if confidence < FALLBACK_CONFIDENCE:
            warnings.append("độ tin cậy thấp, kèm kết quả bộ giải mã DSP")
            if self.fallback:
                fallback_text = profile.render(self._baseline_tokens(x, est.f0_hz).tokens)
        elif confidence < LOW_CONFIDENCE:
            warnings.append("độ tin cậy thấp")
        return DecodeResult(" ".join(w.text for w in words), words,
                            estimate_wpm_from_emissions(emissions, FRAME_RATE), est.f0_hz, est.snr_db,
                            profile.id, scores, warnings, "onnx", fallback_text)

    def decode_file(self, path: str | Path) -> DecodeResult:
        x, sr = load_audio(path)
        return self.decode_array(x, sr)
```

```python
def evaluate(rows: list[dict], engine: str = "baseline", model_path: Path | None = None,
             limit: int | None = None, profile_override: str | None = None, lm: bool = True,
             beam: int = 64, lm_dir: Path | None = None) -> list[dict]:
    decoders: dict[str, Decoder] = {}
    results = []
    for row in rows[:limit] if limit else rows:
        pid = profile_override or row.get("profile", "en")
        if pid not in decoders:
            decoders[pid] = Decoder(pid, engine=engine, model_path=model_path, lm=lm, beam=beam, lm_dir=lm_dir)
        ref = reference_text(row, engine, lm)
        t0 = time.time()
        out = dict(row)
        out["ref"] = ref
        try:
            r = decoders[pid].decode_file(row["path"])
        except Exception as e:  # một file hỏng không được làm mất cả lô
            out.update({
                "hyp": "", "cer": cer(ref, ""), "wer": wer(ref, ""), "wpm_est": None,
                "seconds": time.time() - t0, "warnings": [], "language": None,
                "language_correct": None, "error": f"{type(e).__name__}: {e}",
            })
            results.append(out)
            continue
        out.update({
            "hyp": r.text, "cer": cer(ref, r.text), "wer": wer(ref, r.text), "wpm_est": r.wpm,
            "seconds": time.time() - t0, "warnings": r.warnings, "language": r.language,
            "language_correct": (r.language == row.get("profile")) if profile_override == "auto" else None,
            "error": None,
        })
        results.append(out)
    return results


def language_accuracy(results: list[dict]) -> float | None:
    judged = [r["language_correct"] for r in results if r.get("language_correct") is not None]
    return sum(1 for j in judged if j) / len(judged) if judged else None
```
