```python
"""Chia dải khung thành cửa sổ chồng nhau (dùng chung cho đặc trưng theo cửa sổ và gộp log-prob)."""

from __future__ import annotations

WINDOW_FRAMES = 800       # 4 s ở 200 khung/s (chế độ file)
STEP_FRAMES = 600         # 3 s, chồng 1 s
LIVE_WINDOW_FRAMES = 400  # 2 s (chế độ live)
LIVE_STEP_FRAMES = 200    # 1 s


def frame_windows(n_frames: int, window: int = WINDOW_FRAMES, step: int = STEP_FRAMES) -> list[tuple[int, int]]:
    if window % 2 or step % 2:
        raise ValueError("window và step phải là số chẵn để khớp stride 2 của mô hình")
    if n_frames <= window:
        return [(0, n_frames)]
    starts = list(range(0, n_frames - window, step))
    last = n_frames - window
    last -= last % 2
    if not starts or starts[-1] != last:
        starts.append(last)
    windows = [(s, s + window) for s in starts]
    # Cửa sổ cuối: start làm tròn xuống số chẵn có thể hụt 1 khung so với n_frames; nới điểm cuối
    # tới n_frames để không bỏ sót khung cuối (start <= n_frames - window nên chỉ nới rộng).
    windows[-1] = (windows[-1][0], n_frames)
    return windows
```

```python
"""Cửa sổ trượt cho mô hình ONNX: chạy model trên từng cửa sổ chồng nhau và gộp log-prob,
mỗi khung đầu ra lấy từ cửa sổ có tâm gần nó nhất."""

from __future__ import annotations

import numpy as np

from morsedesk.constants import HOP
from morsedesk.dsp.frontend import compute_features_windowed
from morsedesk.dsp.windowing import (  # noqa: F401  (re-export cho import cũ)
    LIVE_STEP_FRAMES,
    LIVE_WINDOW_FRAMES,
    STEP_FRAMES,
    WINDOW_FRAMES,
    frame_windows,
)


def _merge_pieces(pieces: list[tuple[int, int, np.ndarray]], n_frames: int, stride: int,
                  n_classes: int) -> np.ndarray:
    """pieces: (khung đầu vào bắt đầu, kết thúc, log-prob [T_w', C]) của từng cửa sổ."""
    t_out = (n_frames - 1) // stride + 1 if n_frames > 0 else 0
    out = np.zeros((t_out, n_classes), dtype=np.float32)
    best = np.full(t_out, np.inf)
    filled = np.zeros(t_out, dtype=bool)
    for s, e, lp in pieces:
        g = s // stride + np.arange(lp.shape[0])
        valid = g < t_out
        g, lp = g[valid], lp[valid]
        center = (s + e) / 2.0
        dist = np.abs(g * stride + stride / 2.0 - center)
        better = dist < best[g]
        out[g[better]] = lp[better]
        best[g[better]] = dist[better]
        filled[g[better]] = True
    if not filled.all():
        out[~filled] = np.log(1e-6)
        out[~filled, 0] = 0.0
    return out


def run_chunked(model, feats: np.ndarray, window: int = WINDOW_FRAMES, step: int = STEP_FRAMES) -> np.ndarray:
    """Đặc trưng đã tính cho cả clip (chế độ 'clip'). model: time_stride, output_length, __call__, meta['vocab']."""
    n = feats.shape[0]
    pieces = [(s, e, model(feats[s:e])) for s, e in frame_windows(n, window, step)]
    return _merge_pieces(pieces, n, int(model.time_stride), len(model.meta["vocab"]))


def run_chunked_audio(model, x: np.ndarray, f0_hint: float | None = None, window: int = WINDOW_FRAMES,
                      step: int = STEP_FRAMES) -> tuple[np.ndarray, int]:
    """Chế độ 'window': đặc trưng và tone tính riêng từng cửa sổ. Trả (log-prob gộp, số khung đầu vào)."""
    n_frames = len(x) // HOP
    pieces = [(w.start_frame, w.end_frame, model(w.feats))
              for w in compute_features_windowed(x, f0_hint, window_frames=window, step_frames=step)]
    return _merge_pieces(pieces, n_frames, int(model.time_stride), len(model.meta["vocab"])), n_frames
```

```python
def agc_log(env: np.ndarray, frame_rate: int = FRAME_RATE, tau_s: float = 1.0, init_s: float = 1.0) -> np.ndarray:
    """log(envelope) trừ mức nền trượt, chia độ lệch chuẩn trượt; kẹp về [-6, 6]."""
    if len(env) == 0:
        return np.zeros(0)
    log_env = np.log(env + 1e-6)
    alpha = 1.0 / (tau_s * frame_rate)
    head = log_env[: max(1, min(len(log_env), int(round(init_s * frame_rate))))]
    mean = _ema(log_env, alpha, float(np.mean(head)))
    dev = log_env - mean
    var = _ema(dev * dev, alpha, float(np.mean(dev[: len(head)] ** 2)))
    out = dev / (np.sqrt(np.maximum(var, 0.0)) + 0.1)
    return np.clip(out, -6.0, 6.0)
```

```python
DEFAULT_F0_HZ = 700.0


@dataclass
class Window:
    start_sample: int
    end_sample: int
    f0_hz: float
    feats: np.ndarray
    has_signal: bool

    @property
    def start_frame(self) -> int:
        return self.start_sample // HOP

    @property
    def end_frame(self) -> int:
        return self.end_sample // HOP


def compute_features_windowed(
    x: np.ndarray,
    f0_hint: float | None = None,
    sample_rate: int = SAMPLE_RATE,
    window_frames: int = WINDOW_FRAMES,
    step_frames: int = STEP_FRAMES,
    cfg: FrontendConfig | None = None,
) -> list[Window]:
    """Dò tone và chuẩn hóa đặc trưng trong từng cửa sổ; tone cửa sổ không có tín hiệu lấy từ
    cửa sổ trước, rồi f0_hint, rồi DEFAULT_F0_HZ. Chỉ dùng với sample_rate = SAMPLE_RATE."""
    cfg = cfg or WINDOW_CFG
    n_frames = len(x) // HOP
    if n_frames == 0:
        return []
    x = np.asarray(x, dtype=np.float32)
    windows: list[Window] = []
    carry = f0_hint
    for s, e in frame_windows(n_frames, window_frames, step_frames):
        seg = x[s * HOP:e * HOP]
        est = estimate_tone(seg, sample_rate)
        has_signal = bool(est.has_signal and est.f0_hz is not None)
        if has_signal:
            carry = float(est.f0_hz)
        f0 = carry if carry is not None else DEFAULT_F0_HZ
        windows.append(Window(s * HOP, e * HOP, f0, compute_features(seg, f0, sample_rate, cfg), has_signal))
    return windows


def stitch_windows(windows: list[Window], n_frames: int) -> np.ndarray:
    """Ghép đặc trưng các cửa sổ theo khung: khung t lấy từ cửa sổ có tâm gần t + 0,5 nhất."""
    out = np.zeros((n_frames, FEAT_DIM), dtype=np.float32)
    best = np.full(n_frames, np.inf)
    for w in windows:
        s, e = w.start_frame, w.end_frame
        idx = np.arange(s, min(e, n_frames))
        dist = np.abs(idx + 0.5 - (s + e) / 2.0)
        better = dist < best[idx]
        out[idx[better]] = w.feats[: len(idx)][better]
        best[idx[better]] = dist[better]
    return out
```

```python
"""Huấn luyện CTC on-the-fly bằng SynthGenerator; checkpoint tốt nhất theo token error rate.

Chế độ đặc trưng 'window' (mặc định) tính tone và chuẩn hóa theo cửa sổ 4 s như lúc suy luận;
'clip' là cách của tuần 1–2. Lịch LR: constant hoặc cosine, có warmup. --resume khôi phục optimizer
và bước nếu checkpoint có (checkpoint cũ chỉ nạp trọng số).
"""

from __future__ import annotations

import argparse
import json
import math
import time
from pathlib import Path

import numpy as np
import torch
import torch.nn as nn
from torch.utils.data import DataLoader, IterableDataset, get_worker_info

from morsedesk.constants import HOP
from morsedesk.data.synth import Example, SynthConfig, SynthGenerator
from morsedesk.dsp.frontend import compute_features, compute_features_windowed, stitch_windows
from morsedesk.dsp.tone import estimate_tone
from morsedesk.metrics import edit_distance
from morsedesk.model.ctc import greedy_decode
from morsedesk.model.vocab import VOCAB
from morsedesk.textio import force_utf8_streams
from tools.net_torch import CONFIG_DEFAULT, MorseNet, count_params, output_length

FEATURE_MODES = ("clip", "window")
DEFAULT_FEATURE_MODE = "window"
VAL_SEED_BASE = 2_000_000_000  # seed validation tách hẳn khỏi dải seed train (seed*1e6 + steps*batch)


def example_features(ex: Example, mode: str = DEFAULT_FEATURE_MODE) -> np.ndarray:
    """Đặc trưng như lúc chạy thật. 'window': tone và chuẩn hóa theo cửa sổ, ghép theo khung, tone thật
    làm gợi ý khi cửa sổ không dò được; 'clip': tone toàn cục (rơi về tone thật nếu dò thất bại)."""
    if mode == "window":
        n_frames = len(ex.audio) // HOP
        return stitch_windows(compute_features_windowed(ex.audio, f0_hint=ex.f0_hz), n_frames)
    if mode != "clip":
        raise ValueError(f"feature_mode lạ: {mode}")
    est = estimate_tone(ex.audio)
    f0 = est.f0_hz if est.has_signal and est.f0_hz is not None else ex.f0_hz
    return compute_features(ex.audio, f0)


class SynthDataset(IterableDataset):
    def __init__(self, gen: SynthGenerator, seed_start: int, n_items: int,
                 feature_mode: str = DEFAULT_FEATURE_MODE) -> None:
        self.gen, self.seed_start, self.n_items, self.feature_mode = gen, seed_start, n_items, feature_mode

    def __iter__(self):
        info = get_worker_info()
        wid, nw = (info.id, info.num_workers) if info else (0, 1)
        for k in range(wid, self.n_items, nw):
            ex = self.gen.generate(self.seed_start + k)
            yield (torch.from_numpy(example_features(ex, self.feature_mode)),
                   torch.tensor(ex.labels, dtype=torch.long), ex.tokens)


def collate(batch):
    feats = [b[0] for b in batch]
    t_max = max(f.shape[0] for f in feats)
    padded = torch.zeros(len(feats), t_max, feats[0].shape[1])
    for i, f in enumerate(feats):
        padded[i, : f.shape[0]] = f
    feat_lens = torch.tensor([output_length(f.shape[0]) for f in feats], dtype=torch.long)
    targets = torch.cat([b[1] for b in batch])
    target_lens = torch.tensor([len(b[1]) for b in batch], dtype=torch.long)
    return padded, feat_lens, targets, target_lens, [b[2] for b in batch]


@torch.no_grad()
def token_error_rate(model: nn.Module, items: list[tuple[torch.Tensor, list[str]]]) -> float:
    model.eval()
    errs = total = 0
    for feats, tokens in items:
        logp = model(feats[None])[0].cpu().numpy()
        hyp = [VOCAB.token(i) for i in greedy_decode(logp)]
        errs += edit_distance(tokens, hyp)
        total += len(tokens)
    model.train()
    return errs / max(1, total)


def lr_at(step: int, base_lr: float, total_steps: int, schedule: str = "constant", warmup: int = 0,
          min_ratio: float = 0.05) -> float:
    """LR tại bước step (1-based): warmup tuyến tính, rồi constant hoặc cosine xuống base_lr*min_ratio."""
    if schedule not in ("constant", "cosine"):
        raise ValueError(f"lịch LR lạ: {schedule}")
    if warmup > 0 and step <= warmup:
        return base_lr * step / warmup
    if schedule == "constant":
        return base_lr
    span = max(1, total_steps - warmup)
    progress = min(1.0, max(0.0, (step - warmup) / span))
    return base_lr * (min_ratio + (1.0 - min_ratio) * 0.5 * (1.0 + math.cos(math.pi * progress)))


def save_ckpt(path: Path, model: nn.Module, step: int, val_ter: float, config: dict, opt=None) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    payload = {"state_dict": model.state_dict(), "step": step, "val_ter": val_ter, "config": dict(config)}
    if opt is not None:
        payload["optimizer"] = opt.state_dict()
    torch.save(payload, path)


def load_ckpt(path: Path) -> dict:
    return torch.load(path, map_location="cpu", weights_only=True)


def train_loop(model, loader, opt, ctc, steps, val_items, run_dir: Path, config: dict, log_every=50,
               val_every=500, start_step: int = 0, lr_fn=None, history: list[dict] | None = None) -> dict:
    history = list(history or [])
    best = min([h["val_ter"] for h in history], default=float("inf"))
    step = start_step
    t0 = time.time()
    for feats, feat_lens, targets, target_lens, _ in loader:
        step += 1
        if lr_fn is not None:
            lr = lr_fn(step)
            for g in opt.param_groups:
                g["lr"] = lr
        logp = model(feats)
        loss = ctc(logp.transpose(0, 1), targets, feat_lens, target_lens)
        opt.zero_grad()
        loss.backward()
        nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        opt.step()
        if step % log_every == 0:
            print(f"step {step} loss {loss.item():.4f} lr {opt.param_groups[0]['lr']:.2e} ({time.time() - t0:.0f}s)",
                  flush=True)
        if step % val_every == 0 or step == steps:
            ter = token_error_rate(model, val_items)
            history.append({"step": step, "loss": loss.item(), "val_ter": ter,
                            "lr": opt.param_groups[0]["lr"], "elapsed_s": time.time() - t0})
            save_ckpt(run_dir / "last.pt", model, step, ter, config, opt)
            if ter < best:
                best = ter
                save_ckpt(run_dir / "best.pt", model, step, ter, config, opt)
            print(f"step {step} val TER {100 * ter:.2f}% (best {100 * best:.2f}%)", flush=True)
            (run_dir / "history.json").write_text(json.dumps(history, indent=2), encoding="utf-8")
        if step >= steps:
            break
    (run_dir / "history.json").write_text(json.dumps(history, indent=2), encoding="utf-8")
    return {"best_ter": best, "steps": step, "hours": (time.time() - t0) / 3600}


def main(argv: list[str] | None = None) -> int:
    force_utf8_streams()
    ap = argparse.ArgumentParser(description="Huấn luyện MorseNet bằng dữ liệu tổng hợp")
    ap.add_argument("--name", default="v1")
    ap.add_argument("--steps", type=int, default=20000, help="tổng số bước đích (kể cả bước đã có khi resume)")
    ap.add_argument("--batch", type=int, default=16)
    ap.add_argument("--lr", type=float, default=1e-3)
    ap.add_argument("--lr-schedule", choices=["constant", "cosine"], default="constant")
    ap.add_argument("--warmup", type=int, default=0, help="số bước warmup tuyến tính")
    ap.add_argument("--hidden", type=int, default=CONFIG_DEFAULT["hidden"])
    ap.add_argument("--feature-mode", choices=FEATURE_MODES, default=DEFAULT_FEATURE_MODE)
    ap.add_argument("--seed", type=int, default=1)
    ap.add_argument("--val-clips", type=int, default=200)
    ap.add_argument("--val-every", type=int, default=500)
    ap.add_argument("--workers", type=int, default=0, help="DataLoader workers (0 an toàn trên Windows)")
    ap.add_argument("--no-mp3", action="store_true")
    ap.add_argument("--resume", type=Path, default=None)
    a = ap.parse_args(argv)

    assert a.seed * 1_000_000 + a.steps * a.batch < VAL_SEED_BASE, "seed/steps quá lớn: trùng dải seed validation"

    torch.manual_seed(a.seed)
    torch.set_num_threads(max(1, torch.get_num_threads()))
    config = dict(CONFIG_DEFAULT, hidden=a.hidden, feature_mode=a.feature_mode)
    model = MorseNet(config["feat_dim"], config["n_classes"], config["hidden"])
    opt = torch.optim.AdamW(model.parameters(), lr=a.lr)
    run_dir = Path("runs") / a.name
    start_step = 0
    history: list[dict] = []
    if a.resume:
        ckpt = load_ckpt(a.resume)
        ck_cfg = dict(CONFIG_DEFAULT, **ckpt.get("config", {}))
        if ck_cfg["hidden"] != a.hidden:
            raise SystemExit(f"--hidden {a.hidden} khác checkpoint ({ck_cfg['hidden']})")
        model.load_state_dict(ckpt["state_dict"])
        if "optimizer" in ckpt:
            opt.load_state_dict(ckpt["optimizer"])
            start_step = int(ckpt.get("step", 0))
            hist_path = Path(a.resume).resolve().parent / "history.json"
            if hist_path.exists() and Path(a.resume).resolve().parent == run_dir.resolve():
                history = json.loads(hist_path.read_text(encoding="utf-8"))
            print(f"tiếp tục từ bước {start_step} (optimizer khôi phục)")
        else:
            print("checkpoint cũ: chỉ nạp trọng số, bước bắt đầu 0")
        if ck_cfg.get("feature_mode", "clip") != a.feature_mode:
            print(f"cảnh báo: checkpoint feature_mode={ck_cfg.get('feature_mode', 'clip')}, đang train {a.feature_mode}")
    if start_step >= a.steps:
        raise SystemExit(f"--steps ({a.steps}) phải lớn hơn bước đã có trong checkpoint ({start_step})")
    print(f"tham số: {count_params(model):,} (hidden {a.hidden}, feature_mode {a.feature_mode})")

    gen = SynthGenerator(SynthConfig(mp3_enabled=not a.no_mp3))
    val_items = []
    for i in range(a.val_clips):
        ex = gen.generate(VAL_SEED_BASE + i)
        val_items.append((torch.from_numpy(example_features(ex, a.feature_mode)), ex.tokens))

    ds = SynthDataset(gen, seed_start=a.seed * 1_000_000 + start_step * a.batch,
                      n_items=(a.steps - start_step) * a.batch, feature_mode=a.feature_mode)
    loader = DataLoader(ds, batch_size=a.batch, collate_fn=collate, num_workers=a.workers)
    ctc = nn.CTCLoss(blank=VOCAB.blank_id, zero_infinity=True)
    run_dir.mkdir(parents=True, exist_ok=True)
    (run_dir / "args.json").write_text(json.dumps(vars(a), default=str, indent=2), encoding="utf-8")

    def lr_fn(step: int) -> float:
        return lr_at(step, a.lr, a.steps, a.lr_schedule, a.warmup)

    result = train_loop(model, loader, opt, ctc, a.steps, val_items, run_dir, config, val_every=a.val_every,
                        start_step=start_step, lr_fn=lr_fn, history=history)
    print(json.dumps(result))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
"""Đánh giá một run huấn luyện trên T1: xuất ONNX tạm (models/eval_<name>.onnx), bench greedy và
beam+LM (nếu đủ models/lm/*.bin), ghi runs/<name>/t1_summary.json và in một dòng bảng markdown."""

from __future__ import annotations

import argparse
import json
import math
import time
from pathlib import Path

from morsedesk.bench import aggregate, evaluate, gate, load_rows, pivot_snr_wpm
from morsedesk.lm.store import LMStore
from morsedesk.textio import force_utf8_streams
from tools.export_onnx import export

PROFILE_IDS = ("en", "ru", "kk-cyrl-ru")


def lm_ready(lm_dir: Path | None = None) -> bool:
    store = LMStore(lm_dir)
    return all(store.available(pid) for pid in PROFILE_IDS)


def summarize(results: list[dict]) -> dict:
    ok, cer_gate, n = gate(results)

    def cer_by(key: str) -> dict[str, float]:
        return {k: v["cer"] for k, v in aggregate(results, key).items()}

    return {
        "gate_ok": ok, "gate_cer": cer_gate, "n": n,
        "mean_seconds": sum(r["seconds"] for r in results) / max(1, len(results)),
        "by_profile": cer_by("profile"), "by_channel": cer_by("channel_actual"), "by_keying": cer_by("keying"),
        "by_snr": cer_by("snr"), "by_wpm": cer_by("wpm"),
        "pivot": pivot_snr_wpm(results) if results else "",
    }


def evaluate_run(run_dir: Path, manifest: Path, name: str | None = None, with_lm: bool = True,
                 lm_dir: Path | None = None, models_dir: Path = Path("models"), limit: int | None = None,
                 checkpoint: str = "best.pt") -> dict:
    name = name or run_dir.name
    ckpt = run_dir / checkpoint
    onnx_path = models_dir / f"eval_{name}.onnx"
    export(ckpt, onnx_path, version=f"eval-{name}")
    meta = json.loads(onnx_path.with_suffix(".json").read_text(encoding="utf-8"))
    rows = load_rows(manifest)
    t0 = time.time()
    greedy = evaluate(rows, "onnx", onnx_path, limit, lm=False)
    out = {
        "name": name, "checkpoint": str(ckpt), "onnx": str(onnx_path), "train_step": meta.get("train_step"),
        "val_ter": meta.get("val_ter"), "config": meta.get("config"), "feature_mode": meta.get("feature_mode"),
        "manifest": str(manifest), "n_rows": len(greedy), "greedy": summarize(greedy), "lm": None,
        "eval_seconds": None,
    }
    if with_lm and lm_ready(lm_dir):
        out["lm"] = summarize(evaluate(rows, "onnx", onnx_path, limit, lm=True, lm_dir=lm_dir))
    out["eval_seconds"] = time.time() - t0
    (run_dir / "t1_summary.json").write_text(json.dumps(out, ensure_ascii=False, indent=2), encoding="utf-8")
    return out


def _pct(v: float | None) -> str:
    return "–" if v is None or (isinstance(v, float) and math.isnan(v)) else f"{100 * v:.2f}%"


def markdown_row(s: dict) -> str:
    cfg = s.get("config") or {}
    lm = s.get("lm")
    prof = s["greedy"]["by_profile"]
    per_profile = " / ".join(_pct(prof.get(p)) for p in PROFILE_IDS)
    return (f"| {s['name']} | {cfg.get('hidden')} | {s.get('feature_mode')} | {s.get('train_step')} | "
            f"{_pct(s.get('val_ter'))} | {_pct(s['greedy']['gate_cer'])} | "
            f"{_pct(lm['gate_cer']) if lm else '–'} | {per_profile} |")


def main(argv: list[str] | None = None) -> int:
    force_utf8_streams()
    ap = argparse.ArgumentParser(description="Đánh giá run trên T1")
    ap.add_argument("--run", type=Path, required=True)
    ap.add_argument("--name", default=None)
    ap.add_argument("--manifest", type=Path, default=Path("testdata/generated/t1/manifest.jsonl"))
    ap.add_argument("--no-lm", action="store_true")
    ap.add_argument("--lm-dir", type=Path, default=None)
    ap.add_argument("--limit", type=int, default=None)
    ap.add_argument("--checkpoint", default="best.pt")
    a = ap.parse_args(argv)
    s = evaluate_run(a.run, a.manifest, a.name, not a.no_lm, a.lm_dir, limit=a.limit, checkpoint=a.checkpoint)
    print("greedy:\n" + s["greedy"]["pivot"])
    if s["lm"]:
        print("beam+LM:\n" + s["lm"]["pivot"])
    print(markdown_row(s))
    print(f"-> {a.run / 't1_summary.json'} ({s['eval_seconds']:.0f}s)")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
