"""Kiểm tra model và LM trước khi mở cửa sổ (spec gốc §7: thiếu thì fail ngay)."""

from __future__ import annotations

from pathlib import Path

from morsedesk.api import DEFAULT_MODEL_PATH
from morsedesk.lm.profile import list_profiles, load_profile
from morsedesk.lm.store import default_lm_dir, lm_file_for


def check_models(model_path: Path | None, lm_dir: Path | None, engine: str = "onnx") -> list[Path]:
    """Danh sách file thiếu: ONNX + metadata (nếu engine onnx) rồi các .bin LM theo thứ tự profile."""
    missing: list[Path] = []
    if engine == "onnx":
        onnx = Path(model_path) if model_path else DEFAULT_MODEL_PATH
        for p in (onnx, onnx.with_suffix(".json")):
            if not p.exists():
                missing.append(p)
    lm_root = Path(lm_dir) if lm_dir else default_lm_dir()
    for pid in list_profiles():
        p = lm_file_for(load_profile(pid), lm_root)
        if not p.exists():
            missing.append(p)
    return missing


def missing_models_message(missing: list[Path]) -> str:
    lines = ["Missing model files:"] + [f"  {p}" for p in missing] + [
        "",
        "Export the acoustic model: uv run python tools/export_onnx.py --checkpoint runs/<run>/best.pt --out models/acoustic.onnx",
        "Build the language models: see README (fetch_corpus, normalize_corpus, build_lm.sh, arpa_to_bin) or tools/train_small_lm.py",
        "Or start with --model/--lm-dir pointing to existing files.",
    ]
    return "\n".join(lines)
```

"""Cài đặt người dùng tối thiểu của tuần 4: ngôn ngữ giao diện và thư mục gần nhất (QSettings)."""

from __future__ import annotations

from pathlib import Path

from PySide6.QtCore import QSettings

UI_LANGUAGES = ("en", "ru", "kk")


def default_language(explicit: str | None, saved: str | None, locale_name: str) -> str:
    """explicit hợp lệ > saved hợp lệ > tiền tố locale kk/ru > en."""
    for cand in (explicit, saved):
        if cand in UI_LANGUAGES:
            return cand
    prefix = (locale_name or "").split("_")[0].lower()
    return prefix if prefix in ("kk", "ru") else "en"


class AppSettings:
    def __init__(self, org: str = "MorseDesk", app: str = "MorseDesk", settings: QSettings | None = None) -> None:
        self._s = settings if settings is not None else QSettings(org, app)

    @property
    def ui_language(self) -> str | None:
        v = self._s.value("ui/language", None)
        return str(v) if v else None

    @ui_language.setter
    def ui_language(self, code: str) -> None:
        self._s.setValue("ui/language", code)
        self._s.sync()

    @property
    def last_dir(self) -> str:
        return str(self._s.value("paths/last_dir", str(Path.home())))

    @last_dir.setter
    def last_dir(self, path: str) -> None:
        self._s.setValue("paths/last_dir", path)
        self._s.sync()

    def clear(self) -> None:
        self._s.clear()
        self._s.sync()
```

"""Worker giải mã trong QThread (một Decoder-cache cho cả ứng dụng), spectrogram và phát thử."""

from __future__ import annotations

import logging
import queue
from dataclasses import dataclass
from pathlib import Path

import numpy as np
from PySide6.QtCore import QThread, Signal

from morsedesk.api import DEFAULT_MODEL_PATH, DecodeResult, Decoder
from morsedesk.audio.io import UnsupportedAudio, audio_info, load_audio
from morsedesk.constants import HOP, SAMPLE_RATE
from morsedesk.dsp.stft import stft_power
from morsedesk.lm.store import LMStore

log = logging.getLogger(__name__)

MAX_SECONDS = 20 * 60
MAX_COLUMNS = 4000
MAX_FREQ_HZ = 2000.0
DB_FLOOR = -60.0


@dataclass
class Spectrogram:
    power_db: np.ndarray   # [n_bins, n_cols] float32, dB tương đối trong [DB_FLOOR, 0]
    freqs_hz: np.ndarray   # [n_bins]
    times_s: np.ndarray    # [n_cols] giữa mỗi cột

    @classmethod
    def from_audio(cls, x: np.ndarray, sample_rate: int = SAMPLE_RATE,
                   max_columns: int = MAX_COLUMNS) -> Spectrogram:
        x = np.asarray(x, dtype=np.float32)
        n_frames = len(x) // HOP
        if n_frames == 0:
            return cls(np.zeros((0, 0), np.float32), np.zeros(0), np.zeros(0))
        power, freqs = stft_power(x, sample_rate=sample_rate)
        keep = freqs <= MAX_FREQ_HZ
        p = power[keep, :]
        db = 10.0 * np.log10(p + 1e-12)
        db = db - float(db.max())
        db = np.clip(db, DB_FLOOR, 0.0).astype(np.float32)
        times = (np.arange(db.shape[1]) * HOP + HOP / 2.0) / sample_rate
        if db.shape[1] > max_columns:
            factor = int(np.ceil(db.shape[1] / max_columns))
            n_pool = db.shape[1] // factor
            pooled = db[:, : n_pool * factor].reshape(db.shape[0], n_pool, factor).max(axis=2)
            times = times[: n_pool * factor].reshape(n_pool, factor).mean(axis=1)
            db = pooled
        return cls(db, freqs[keep].astype(np.float64), times.astype(np.float64))


@dataclass(frozen=True)
class DecodeRequest:
    path: Path
    profile_id: str
    use_lm: bool


class DecoderWorker(QThread):
    """Sở hữu mọi Decoder; UI chỉ gửi DecodeRequest và nhận signal."""

    result_ready = Signal(object, object, str)   # DecodeResult, Spectrogram, path
    failed = Signal(str)
    busy = Signal(bool)

    def __init__(self, engine: str, model_path: Path | None, lm_dir: Path | None, parent=None) -> None:
        super().__init__(parent)
        self.engine = engine
        self.model_path = Path(model_path) if model_path else None
        self.lm_dir = Path(lm_dir) if lm_dir else None
        self._queue: queue.Queue = queue.Queue(maxsize=1)
        self._decoders: dict[tuple[str, bool], Decoder] = {}
        self._acoustic = None
        self._store: LMStore | None = None

    # ---- gọi từ UI thread ----
    def request(self, req: DecodeRequest) -> None:
        while True:                                   # yêu cầu mới thay yêu cầu đang chờ
            try:
                self._queue.get_nowait()
            except queue.Empty:
                break
        self._queue.put(req)

    def stop(self) -> None:
        self.request(None)  # type: ignore[arg-type]
        self.wait(10000)

    # ---- chạy trong thread ----
    def _decoder(self, req: DecodeRequest) -> Decoder:
        key = (req.profile_id, req.use_lm)
        if key not in self._decoders:
            if self._store is None:
                self._store = LMStore(self.lm_dir)
            if self.engine == "onnx" and self._acoustic is None:
                from morsedesk.model.onnx_runner import OnnxAcousticModel

                self._acoustic = OnnxAcousticModel(self.model_path or DEFAULT_MODEL_PATH)
            self._decoders[key] = Decoder(req.profile_id, engine=self.engine, model_path=self.model_path,
                                          lm=req.use_lm, lm_store=self._store, acoustic_model=self._acoustic)
        return self._decoders[key]

    def _handle(self, req: DecodeRequest) -> None:
        path = Path(req.path)
        if not path.exists():
            raise FileNotFoundError(f"file not found: {path}")
        try:
            _, _, seconds = audio_info(path)
        except Exception as e:  # soundfile không đọc được (định dạng lạ, file hỏng)
            raise UnsupportedAudio(f"cannot read {path.name}: {e}") from e
        if seconds > MAX_SECONDS:
            raise ValueError(f"file is {seconds / 60:.1f} min long; this version supports up to 20 min")
        decoder = self._decoder(req)
        x, sr = load_audio(path)
        result: DecodeResult = decoder.decode_array(x, sr)
        spec = Spectrogram.from_audio(x, SAMPLE_RATE)
        self.result_ready.emit(result, spec, str(path))

    def run(self) -> None:
        while True:
            req = self._queue.get()
            if req is None:
                return
            self.busy.emit(True)
            try:
                self._handle(req)
            except (UnsupportedAudio, FileNotFoundError, ValueError) as e:
                self.failed.emit(str(e))
            except Exception as e:  # không để worker chết vì một file lạ
                log.exception("decode failed")
                self.failed.emit(f"internal error: {type(e).__name__}: {e}")
            finally:
                self.busy.emit(False)


class Player:
    """Phát thử qua sounddevice (không chặn); không có thiết bị thì trả thông điệp lỗi."""

    def play(self, x: np.ndarray, sample_rate: int) -> str | None:
        try:
            import sounddevice as sd

            sd.stop()
            sd.play(np.asarray(x, dtype=np.float32), sample_rate)
            return None
        except Exception as e:  # PortAudioError, OSError khi thiếu thiết bị/thư viện
            return f"cannot play audio: {e}"

    def stop(self) -> None:
        try:
            import sounddevice as sd

            sd.stop()
        except Exception:
            pass

"""FlowLayout: xếp widget theo hàng, xuống dòng khi hết chiều rộng (theo ví dụ Qt)."""

from __future__ import annotations

from PySide6.QtCore import QPoint, QRect, QSize, Qt
from PySide6.QtWidgets import QLayout, QSizePolicy


class FlowLayout(QLayout):
    def __init__(self, parent=None, margin: int = 0, spacing: int = 4) -> None:
        super().__init__(parent)
        self.setContentsMargins(margin, margin, margin, margin)
        self._spacing = spacing
        self._items = []

    def addItem(self, item) -> None:
        self._items.append(item)

    def count(self) -> int:
        return len(self._items)

    def itemAt(self, index: int):
        return self._items[index] if 0 <= index < len(self._items) else None

    def takeAt(self, index: int):
        return self._items.pop(index) if 0 <= index < len(self._items) else None

    def expandingDirections(self):
        return Qt.Orientation(0)

    def hasHeightForWidth(self) -> bool:
        return True

    def heightForWidth(self, width: int) -> int:
        return self._do_layout(QRect(0, 0, width, 0), test_only=True)

    def setGeometry(self, rect: QRect) -> None:
        super().setGeometry(rect)
        self._do_layout(rect, test_only=False)

    def sizeHint(self) -> QSize:
        return self.minimumSize()

    def minimumSize(self) -> QSize:
        size = QSize()
        for item in self._items:
            size = size.expandedTo(item.minimumSize())
        m = self.contentsMargins()
        return size + QSize(m.left() + m.right(), m.top() + m.bottom())

    def clear(self) -> None:
        while self._items:
            item = self._items.pop()
            w = item.widget()
            if w is not None:
                w.setParent(None)
                w.deleteLater()

    def _do_layout(self, rect: QRect, test_only: bool) -> int:
        x, y = rect.x(), rect.y()
        line_height = 0
        for item in self._items:
            w = item.widget()
            if w is not None and w.isHidden():
                continue
            hint = item.sizeHint()
            next_x = x + hint.width() + self._spacing
            if next_x - self._spacing > rect.right() and line_height > 0:
                x = rect.x()
                y += line_height + self._spacing
                next_x = x + hint.width() + self._spacing
                line_height = 0
            if not test_only:
                item.setGeometry(QRect(QPoint(x, y), hint))
            x = next_x
            line_height = max(line_height, hint.height())
        return y + line_height - rect.y()


_ = QSizePolicy  # giữ import cho type checker của Qt (không dùng trực tiếp)

from __future__ import annotations

from PySide6.QtCore import QEvent, Qt, Signal
from PySide6.QtWidgets import QLabel, QWidget

from morsedesk.api import FALLBACK_CONFIDENCE, Word
from morsedesk.app.widgets.flow_layout import FlowLayout

WORD_GOOD = 0.8
COLOR_GOOD, COLOR_MID, COLOR_LOW = "#c8f0c8", "#fff2b3", "#f7c6c6"


def confidence_color(c: float) -> str:
    if c >= WORD_GOOD:
        return COLOR_GOOD
    if c >= FALLBACK_CONFIDENCE:
        return COLOR_MID
    return COLOR_LOW


class WordLabel(QLabel):
    clicked = Signal(float)

    def __init__(self, word: Word, parent=None) -> None:
        super().__init__(word.text, parent)
        self.word = word
        self.setStyleSheet(f"QLabel {{ background:{confidence_color(word.confidence)}; padding:3px 6px; "
                           "border-radius:4px; font-size:16pt; }")
        self.setCursor(Qt.CursorShape.PointingHandCursor)
        self.retranslate()

    def retranslate(self) -> None:
        self.setToolTip(self.tr("confidence {conf:.2f} · {start:.2f}–{end:.2f} s").format(
            conf=self.word.confidence, start=self.word.start_s, end=self.word.end_s))

    def mousePressEvent(self, event) -> None:
        if event.button() == Qt.MouseButton.LeftButton:
            self.clicked.emit(self.word.start_s)
        super().mousePressEvent(event)


class WordView(QWidget):
    word_clicked = Signal(float)

    def __init__(self, parent=None) -> None:
        super().__init__(parent)
        self._layout = FlowLayout(self, margin=6, spacing=6)
        self._labels: list[WordLabel] = []
        self._placeholder = QLabel(self)
        self._placeholder.setStyleSheet("color: gray;")
        self._layout.addWidget(self._placeholder)
        self.retranslate()

    def set_placeholder(self, text: str) -> None:
        self._placeholder.setText(text)

    def set_words(self, words: list[Word]) -> None:
        self.clear()
        self._placeholder.setVisible(not words)
        for w in words:
            lb = WordLabel(w, self)
            lb.clicked.connect(self.word_clicked.emit)
            self._layout.addWidget(lb)
            self._labels.append(lb)
        self._layout.invalidate()

    def clear(self) -> None:
        for lb in self._labels:
            self._layout.removeWidget(lb)
            lb.setParent(None)
            lb.deleteLater()
        self._labels = []
        self._placeholder.setVisible(True)

    def labels(self) -> list[WordLabel]:
        return list(self._labels)

    def retranslate(self) -> None:
        self._placeholder.setText(self.tr("Open an audio file to decode."))
        for lb in self._labels:
            lb.retranslate()

    def changeEvent(self, event) -> None:
        if event.type() == QEvent.Type.LanguageChange:
            self.retranslate()

"""Spectrogram 0–2 kHz bằng pyqtgraph, vạch tone, vạch đầu từ, con trỏ thời gian."""

from __future__ import annotations

import numpy as np
import pyqtgraph as pg
from PySide6.QtCore import QRectF

from morsedesk.app.worker import DB_FLOOR, MAX_FREQ_HZ, Spectrogram


class SpectrogramView(pg.PlotWidget):
    def __init__(self, parent=None) -> None:
        super().__init__(parent, background="k")
        self.setLabel("bottom", "s")
        self.setLabel("left", "Hz")
        self.setYRange(0, MAX_FREQ_HZ, padding=0)
        self.setMouseEnabled(x=True, y=False)
        self.image = pg.ImageItem()
        self.image.setLookupTable(pg.colormap.get("viridis").getLookupTable())
        self.addItem(self.image)
        self.tone_line = pg.InfiniteLine(angle=0, pen=pg.mkPen("#ff5555", width=2))
        self.tone_line.setVisible(False)
        self.addItem(self.tone_line)
        self.cursor_line = pg.InfiniteLine(angle=90, pen=pg.mkPen("#ffffff", width=1))
        self.cursor_line.setVisible(False)
        self.addItem(self.cursor_line)
        self.word_lines: list[pg.InfiniteLine] = []

    def set_spectrogram(self, spec: Spectrogram | None) -> None:
        if spec is None or spec.power_db.size == 0:
            self.image.clear()
            return
        self.image.setImage(np.ascontiguousarray(spec.power_db.T), autoLevels=False, levels=(DB_FLOOR, 0.0))
        t_end = float(spec.times_s[-1] + (spec.times_s[1] - spec.times_s[0] if len(spec.times_s) > 1 else 0.02))
        f_end = float(spec.freqs_hz[-1])
        self.image.setRect(QRectF(0.0, 0.0, t_end, f_end))
        self.setXRange(0.0, t_end, padding=0)

    def set_tone(self, hz: float | None) -> None:
        if hz is None:
            self.tone_line.setVisible(False)
        else:
            self.tone_line.setValue(float(hz))
            self.tone_line.setVisible(True)

    def set_word_marks(self, starts_s: list[float]) -> None:
        for ln in self.word_lines:
            self.removeItem(ln)
        self.word_lines = []
        for s in starts_s:
            ln = pg.InfiniteLine(angle=90, pos=float(s), pen=pg.mkPen((255, 255, 255, 80), width=1))
            self.addItem(ln)
            self.word_lines.append(ln)

    def set_cursor(self, t_s: float | None) -> None:
        if t_s is None:
            self.cursor_line.setVisible(False)
        else:
            self.cursor_line.setValue(float(t_s))
            self.cursor_line.setVisible(True)

    def clear(self) -> None:
        self.image.clear()
        self.set_tone(None)
        self.set_word_marks([])
        self.set_cursor(None)

        page = QWidget(self)

    # ---- i18n ----
    def set_language(self, code: str) -> bool:
        if code not in UI_LANGUAGES:
            return False
        app = QCoreApplication.instance()
        if self._translator is not None:
            app.removeTranslator(self._translator)
            self._translator = None

            tr = QTranslator(self)
            ok = tr.load(str(self.i18n_dir / f"morsedesk_{code}.qm"))

                app.installTranslator(tr)
                self._translator = tr

        if idx >= 0 and self.language_combo.currentIndex() != idx:
            self.language_combo.blockSignals(True)
            self.language_combo.setCurrentIndex(idx)
            self.language_combo.blockSignals(False)
        self.retranslate()
        return ok

"""Entry `morsedesk-gui`: tham số, log, kiểm model, QApplication, cửa sổ chính."""

from __future__ import annotations

import argparse
import logging
import sys
from logging.handlers import RotatingFileHandler
from pathlib import Path

from PySide6.QtCore import QLocale, QStandardPaths
from PySide6.QtWidgets import QApplication, QMessageBox

from morsedesk.app.main_window import MainWindow

I18N_DIR = Path(__file__).resolve().parent / "i18n"
log = logging.getLogger("morsedesk.app")


def build_parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(prog="morsedesk-gui", description="MorseDesk GUI")
    p.add_argument("--model", type=Path, default=None, help="ONNX model (default models/acoustic.onnx)")
    p.add_argument("--lm-dir", type=Path, default=None, help="language model directory (default models/lm)")
    p.add_argument("--engine", choices=["onnx", "baseline"], default="onnx")
    p.add_argument("--lang", choices=["en", "ru", "kk"], default=None, help="UI language override")
    return p


def setup_logging() -> None:
    base = QStandardPaths.writableLocation(QStandardPaths.StandardLocation.AppDataLocation)
    log_dir = Path(base) / "logs"
    try:
        log_dir.mkdir(parents=True, exist_ok=True)
        handler = RotatingFileHandler(log_dir / "morsedesk.log", maxBytes=1_000_000, backupCount=3, encoding="utf-8")
        handler.setFormatter(logging.Formatter("%(asctime)s %(levelname)s %(name)s: %(message)s"))
        logging.getLogger().addHandler(handler)
        logging.getLogger().setLevel(logging.INFO)
    except OSError:
        logging.basicConfig(level=logging.INFO)


def show_missing_dialog(message: str) -> None:
    app = QApplication.instance() or QApplication(sys.argv[:1])
    QMessageBox.critical(None, "MorseDesk", message)
    _ = app


def load_settings() -> AppSettings:
    return AppSettings()


def create_app(args: argparse.Namespace) -> tuple[QApplication, MainWindow]:
    app = QApplication.instance() or QApplication(sys.argv[:1])
    app.setApplicationName("MorseDesk")
    app.setOrganizationName("MorseDesk")
    settings = load_settings()
    worker = DecoderWorker(engine=args.engine, model_path=args.model, lm_dir=args.lm_dir)
    worker.start()
    window = MainWindow(worker, settings, I18N_DIR)
    window.set_language(default_language(args.lang, settings.ui_language, QLocale.system().name()))
    return app, window


def main(argv: list[str] | None = None) -> int:
    force_utf8_streams()
    args = build_parser().parse_args(argv)
    setup_logging()
    missing = check_models(args.model, args.lm_dir, args.engine)
    if missing:
        msg = missing_models_message(missing)
        log.error(msg)
        show_missing_dialog(msg)
        return 2

    def excepthook(exc_type, exc, tb):
        log.error("unhandled exception", exc_info=(exc_type, exc, tb))
        sys.__excepthook__(exc_type, exc, tb)

    sys.excepthook = excepthook
    app, window = create_app(args)
    window.show()
    return int(app.exec())


if __name__ == "__main__":
    raise SystemExit(main())

"""Sinh .ts bằng pyside6-lupdate, điền bản dịch từ translations.json, biên dịch .qm bằng pyside6-lrelease.

Dùng: uv run python tools/build_i18n.py            # sinh lại .ts/.qm
      uv run python tools/build_i18n.py --check    # chỉ kiểm bảng dịch có đủ chuỗi tr() không (không cần Qt tools)
"""

from __future__ import annotations

import argparse
import json
import re
import shutil
import subprocess
import sys
import xml.etree.ElementTree as ET  # .ts do pyside6-lupdate của chính dự án sinh (đầu vào tin cậy, không cần defusedxml)
from pathlib import Path

from morsedesk.textio import force_utf8_streams

ROOT = Path(__file__).resolve().parents[1]
APP_DIR = ROOT / "morsedesk" / "app"
I18N_DIR = APP_DIR / "i18n"
LANGS = ("ru", "kk")
SOURCES = sorted(p for p in APP_DIR.rglob("*.py"))
_TR_RE = re.compile(r'self\.tr\("((?:[^"\\]|\\.)*)"\)')


def tr_literals(paths: list[Path]) -> set[str]:
    out: set[str] = set()
    for p in paths:
        for m in _TR_RE.finditer(p.read_text(encoding="utf-8")):
            out.add(m.group(1).encode("utf-8").decode("unicode_escape") if "\\" in m.group(1) else m.group(1))
    return out


def fill_ts(ts_path: Path, table: dict[str, str]) -> int:
    tree = ET.parse(ts_path)
    left = 0
    for msg in tree.getroot().iter("message"):
        src = msg.findtext("source") or ""
        tr = msg.find("translation")
        if tr is None:
            tr = ET.SubElement(msg, "translation")
        if src in table:
            tr.text = table[src]
            tr.attrib.pop("type", None)
        else:
            tr.set("type", "unfinished")
            left += 1
    tree.write(ts_path, encoding="utf-8", xml_declaration=True)
    text = ts_path.read_text(encoding="utf-8")
    if "<!DOCTYPE TS>" not in text:
        text = text.replace("?>", "?>\n<!DOCTYPE TS>", 1)
        ts_path.write_text(text, encoding="utf-8")
    return left


def _tool(name: str) -> str:
    exe = shutil.which(name) or shutil.which(name + ".exe")
    if exe is None:
        scripts = Path(sys.executable).parent
        for cand in (scripts / name, scripts / f"{name}.exe"):
            if cand.exists():
                return str(cand)
        raise SystemExit(f"không tìm thấy {name}; cài extra gui: uv sync --all-extras")
    return exe


def main(argv: list[str] | None = None) -> int:
    force_utf8_streams()
    ap = argparse.ArgumentParser(description="Dựng bản dịch Qt cho GUI")
    ap.add_argument("--check", action="store_true", help="chỉ kiểm bảng dịch đủ chuỗi")
    a = ap.parse_args(argv)
    table = json.loads((I18N_DIR / "translations.json").read_text(encoding="utf-8"))
    literals = tr_literals(SOURCES)
    missing = {lang: sorted(s for s in literals if s not in table[lang]) for lang in LANGS}
    for lang, miss in missing.items():
        for s in miss:
            print(f"{lang}: thiếu {s!r}")
    if a.check:
        return 1 if any(missing.values()) else 0
    ts_files = [I18N_DIR / f"morsedesk_{lang}.ts" for lang in LANGS]
    subprocess.run([_tool("pyside6-lupdate"), *map(str, SOURCES), "-no-obsolete", "-ts", *map(str, ts_files)],
                   check=True)
    for lang, ts in zip(LANGS, ts_files, strict=True):
        left = fill_ts(ts, table[lang])
        subprocess.run([_tool("pyside6-lrelease"), str(ts), "-qm", str(ts.with_suffix(".qm"))], check=True)
        print(f"{lang}: {ts.name} -> {ts.with_suffix('.qm').name}, chưa dịch {left}")
    return 1 if any(missing.values()) else 0


if __name__ == "__main__":
    raise SystemExit(main())

## Lệch so với spec tuần 4 đã biết khi viết kế hoạch

- Nghe thử dùng `Player` (bọc `sounddevice.play`, không chặn) thay `EncoderWorker(QThread)`: tạo audio mất < 100 ms, không cần thread; giữ tên `worker.py` cho `DecoderWorker`, `Spectrogram`, `Player`.

- Cửa sổ chính không có `sys.excepthook` hiện banner (spec §2.2 nói banner đỏ "Lỗi nội bộ"): excepthook ghi log rồi gọi hook mặc định; banner nội bộ để tuần 5 khi có live (nguồn lỗi bất đồng bộ nhiều hơn). Lỗi trong worker vẫn tới banner qua `failed`.
