```python
"""Thu âm thanh trực tiếp: liệt kê thiết bị và đẩy khối PortAudio vào hàng đợi.

Module thuộc lõi nên KHÔNG import Qt. `sounddevice` được import trễ trong hàm: máy đích
air-gap có thể thiếu `libportaudio2`, khi đó `import sounddevice` ném `OSError` ngay lúc
nạp module và sẽ kéo sập cả CLI lẫn GUI nếu import ở đầu file.
"""

from __future__ import annotations

import logging
import queue
from dataclasses import dataclass

import numpy as np

from morsedesk.constants import SAMPLE_RATE

log = logging.getLogger(__name__)

BLOCK_MS = 100          # độ dài một khối thu; 100 ms = 800 mẫu ở 8 kHz
MAX_BLOCKS = 50         # hàng đợi 5 giây; tràn thì bỏ khối MỚI và tăng bộ đếm


class CaptureError(RuntimeError):
    """Không liệt kê hoặc không mở được thiết bị thu (thiếu PortAudio, thiết bị bận/biến mất)."""


@dataclass(frozen=True)
class AudioDevice:
    index: int
    name: str
    channels: int
    samplerate: float


def _sounddevice():
    try:
        import sounddevice as sd
    except Exception as e:      # ImportError (không cài) hoặc OSError (thiếu libportaudio2)
        raise CaptureError(f"sounddevice unavailable: {type(e).__name__}: {e}") from e
    return sd


def _devices(key: str) -> list[AudioDevice]:
    sd = _sounddevice()
    try:
        raw = sd.query_devices()
    except Exception as e:
        raise CaptureError(f"cannot query audio devices: {type(e).__name__}: {e}") from e
    out: list[AudioDevice] = []
    for i, d in enumerate(raw):
        channels = int(d.get(key, 0) or 0)
        if channels <= 0:
            continue
        out.append(AudioDevice(
            index=int(d.get("index", i)),
            name=str(d.get("name", f"device {i}")),
            channels=channels,
            samplerate=float(d.get("default_samplerate", SAMPLE_RATE) or SAMPLE_RATE),
        ))
    return out


def list_input_devices() -> list[AudioDevice]:
    """Thiết bị có kênh thu; ném `CaptureError` nếu không dùng được sounddevice."""
    return _devices("max_input_channels")


def list_output_devices() -> list[AudioDevice]:
    """Thiết bị có kênh phát; ném `CaptureError` nếu không dùng được sounddevice."""
    return _devices("max_output_channels")


class AudioCapture:
    """Bọc `sounddevice.InputStream`: callback đẩy khối mono float32 vào `queue.Queue`.

    Hàng đợi đầy nghĩa là bên tiêu thụ (LiveWorker) không theo kịp: bỏ khối vừa tới và
    tăng `dropped`. Tuyệt đối không chặn trong callback — PortAudio gọi nó từ thread
    thời gian thực, chặn ở đó làm méo âm và có thể treo driver.
    """

    def __init__(self, device: int | None = None, block_ms: int = BLOCK_MS,
                 sample_rate: int = SAMPLE_RATE, max_blocks: int = MAX_BLOCKS) -> None:
        self.device = device
        self.block_ms = int(block_ms)
        self.sample_rate = int(sample_rate)
        self.blocksize = int(self.sample_rate * self.block_ms / 1000)
        self._queue: queue.Queue = queue.Queue(maxsize=int(max_blocks))
        self._dropped = 0
        self._stream = None

    @property
    def dropped(self) -> int:
        return self._dropped

    @property
    def pending(self) -> int:
        return self._queue.qsize()

    def _callback(self, indata, frames, time_info, status) -> None:
        data = np.asarray(indata, dtype=np.float32)
        mono = data.mean(axis=1) if data.ndim > 1 and data.shape[1] > 1 else data.reshape(-1)
        try:
            self._queue.put_nowait(np.array(mono, dtype=np.float32, copy=True))
        except queue.Full:
            self._dropped += 1

    def start(self) -> None:
        """Mở và chạy luồng thu; gọi lại khi đã chạy là vô hại."""
        if self._stream is not None:
            return
        sd = _sounddevice()
        try:
            stream = sd.InputStream(
                samplerate=self.sample_rate, blocksize=self.blocksize, device=self.device,
                channels=1, dtype="float32", callback=self._callback,
            )
            stream.start()
        except Exception as e:
            raise CaptureError(f"cannot open input device {self.device}: {type(e).__name__}: {e}") from e
        self._stream = stream

    def stop(self) -> None:
        """Dừng và đóng luồng; an toàn khi chưa start hoặc start đã hỏng."""
        stream, self._stream = self._stream, None
        if stream is None:
            return
        try:
            stream.stop()
            stream.close()
        except Exception:
            log.warning("closing input stream failed", exc_info=True)

    def get(self, timeout: float = 0.2) -> np.ndarray | None:
        """Khối tiếp theo, hoặc `None` khi hết `timeout` (để vòng lặp kiểm cờ dừng)."""
        try:
            return self._queue.get(timeout=timeout)
        except queue.Empty:
            return None

    def __enter__(self) -> AudioCapture:
        self.start()
        return self

    def __exit__(self, exc_type, exc, tb) -> None:
        self.stop()
```

```python
class StreamDecoder:
    """Giải mã theo luồng cho chế độ live và cho file dài (spec tuần 5 §4.1).

    Giữ bộ đệm trượt dài `window_s` giây. Mỗi khi nhận thêm `decode_every_s` giây audio thì
    giải mã lại **cả cửa sổ đuôi** và chỉ phát ra những từ đã ổn định: kết thúc trước mép
    chồng lấn (`overlap_s` giây cuối) và bắt đầu sau từ cuối đã phát. Nhờ vậy văn bản chỉ
    nối thêm, không bao giờ tự viết lại — điều kiện duy nhất là không có "từ" nào dài hơn
    `window_s - overlap_s` giây (ở 10 WPM tương đương 9 ký tự liền không có khoảng trắng).
    """

    def __init__(
        self, profile: str | Profile = "en", engine: str = "baseline", *,
        window_s: float = LIVE_WINDOW_S, overlap_s: float = LIVE_OVERLAP_S,
        decode_every_s: float = LIVE_DECODE_EVERY_S, sample_rate: int = SAMPLE_RATE,
        decoder: Decoder | None = None, model_path: Path | None = None, lm: bool = True,
        lm_dir: Path | None = None, acoustic_model: AcousticModel | None = None,
        lm_store: LMStore | None = None, fallback: bool = False,
    ) -> None:
        """`decoder`: dùng lại một Decoder có sẵn (GUI chia sẻ ONNX session và LMStore)."""
        self.decoder = decoder if decoder is not None else Decoder(
            profile, engine=engine, model_path=model_path, fallback=fallback, lm=lm,
            lm_dir=lm_dir, acoustic_model=acoustic_model, lm_store=lm_store,
        )
        self.sample_rate = int(sample_rate)
        self.window = int(window_s * self.sample_rate)
        self.overlap = int(overlap_s * self.sample_rate)
        self.hop = max(1, int(decode_every_s * self.sample_rate))
        self.min_samples = self.sample_rate // 2      # dưới 0,5 s thì chưa dò nổi tone
        self.reset()

    def reset(self) -> None:
        """Về trạng thái ban đầu (đổi thiết bị, đổi profile, bắt đầu file mới)."""
        self._buf = np.zeros(0, dtype=np.float32)
        self._dropped_samples = 0        # số mẫu đã cắt khỏi đầu bộ đệm
        self._pending = 0                # số mẫu mới kể từ lần giải mã gần nhất
        self._emitted_end = 0.0          # giây tuyệt đối: hết từ cuối đã phát ra
        self._warn_seen: set[str] = set()
        self.words: list[Word] = []
        self.warnings: list = []
        self.last_result: DecodeResult | None = None

    @property
    def seconds_fed(self) -> float:
        return (self._dropped_samples + len(self._buf)) / self.sample_rate

    @property
    def text(self) -> str:
        return " ".join(w.text for w in self.words)

    def feed(self, block: np.ndarray) -> list[Word]:
        """Nạp một khối audio; trả về những từ vừa ổn định (rỗng nếu chưa tới lượt giải mã)."""
        b = np.asarray(block, dtype=np.float32).reshape(-1)
        if b.size:
            self._buf = np.concatenate([self._buf, b])
            self._pending += int(b.size)
        if len(self._buf) > self.window:
            cut = len(self._buf) - self.window
            self._buf = self._buf[cut:]
            self._dropped_samples += cut
        if self._pending < self.hop:
            return []
        self._pending = 0
        return self._decode(self.overlap)

    def flush(self) -> list[Word]:
        """Kết thúc luồng: giải mã nốt phần đuôi, không giữ lại vùng chồng lấn nữa."""
        self._pending = 0
        return self._decode(0)

    def result(self) -> DecodeResult:
        """Gộp toàn bộ từ đã phát thành một `DecodeResult` như chế độ file."""
        last = self.last_result
        return DecodeResult(
            self.text, list(self.words),
            last.wpm if last else None,
            last.tone_hz if last else None,
            last.snr_db if last else None,
            last.language if last else "",
            dict(last.language_scores) if last else {},
            list(self.warnings),
            self.decoder.engine,
            None,
        )

    def _decode(self, guard: int) -> list[Word]:
        if len(self._buf) < self.min_samples:
            return []
        result = self.decoder.decode_array(self._buf, self.sample_rate)
        self.last_result = result
        for w in result.warnings:
            key = repr(w)
            if key not in self._warn_seen:
                self._warn_seen.add(key)
                self.warnings.append(w)
        offset = self._dropped_samples / self.sample_rate
        edge = (self._dropped_samples + len(self._buf) - guard) / self.sample_rate
        new: list[Word] = []
        for w in result.words:
            start, end = offset + w.start_s, offset + w.end_s
            # Bỏ từ còn trong vùng chồng lấn (chưa chắc), và bỏ mảnh vỡ ở mép trái cửa sổ:
            # mảnh vỡ luôn bắt đầu TRƯỚC chỗ đã phát tới, nên điều kiện start là đủ và chặt.
            if end > edge + 1e-9 or start < self._emitted_end - 1e-3:
                continue
            new.append(Word(w.text, start, end, w.confidence))
            self._emitted_end = end
        self.words += new
        return new
```

```python
@dataclass
class Spectrogram:
    power_db: np.ndarray   # [n_bins, n_cols] float32, dB tương đối trong [DB_FLOOR, 0]
    freqs_hz: np.ndarray   # [n_bins]
    times_s: np.ndarray    # [n_cols] giữa mỗi cột
    peak_power: float = 0.0  # công suất STFT lớn nhất trong dải giữ lại (mốc dB của mảnh này)

    @classmethod
    def from_audio(
        cls, x: np.ndarray, sample_rate: int = SAMPLE_RATE, max_columns: int = MAX_COLUMNS,
        ref_power: float | None = None,
    ) -> Spectrogram:
        """`ref_power`: mốc 0 dB dùng chung giữa nhiều mảnh (waterfall live, file theo luồng).

        Để `None` thì mỗi mảnh tự lấy đỉnh của chính nó làm mốc — đúng cho một clip trọn vẹn,
        nhưng sai cho luồng: khối im lặng sẽ bị kéo lên 0 dB và waterfall nhấp nháy.
        """
        x = np.asarray(x, dtype=np.float32)
        n_frames = len(x) // HOP
        if n_frames == 0:
            return cls(np.zeros((0, 0), np.float32), np.zeros(0), np.zeros(0), 0.0)
        power, freqs = stft_power(x, sample_rate=sample_rate)
        keep = freqs <= MAX_FREQ_HZ
        p = power[keep, :]
        peak = float(p.max()) if p.size else 0.0
        ref = peak if ref_power is None else float(ref_power)
        db = 10.0 * np.log10(p + 1e-12) - 10.0 * np.log10(ref + 1e-12)
        db = np.clip(db, DB_FLOOR, 0.0).astype(np.float32)
        times = (np.arange(db.shape[1]) * HOP + HOP / 2.0) / sample_rate
        if db.shape[1] > max_columns:
            factor = int(np.ceil(db.shape[1] / max_columns))
            n_pool = db.shape[1] // factor
            pooled = db[:, : n_pool * factor].reshape(db.shape[0], n_pool, factor).max(axis=2)
            times = times[: n_pool * factor].reshape(n_pool, factor).mean(axis=1)
            db = pooled
        return cls(db, freqs[keep].astype(np.float64), times.astype(np.float64), peak)
```

```python
"""Thread thu-và-giải-mã trực tiếp: thread thứ hai bên cạnh `DecoderWorker`.

Vì sao không `moveToThread`: `DecoderWorker` là QThread có vòng `queue.get()` chặn, không có
event loop, nên không nhận được queued slot call. Chuyển nó sang `moveToThread` + `QEventLoop`
đồng nghĩa viết lại phần đã kiểm chứng của tuần 4. Thay vào đó `LiveWorker` là thread thứ hai,
cũng một vòng `queue.get()` chặn, độc lập hoàn toàn: hai thread không gọi slot của nhau, chỉ
phát signal lên UI thread (spec tuần 5 §4.1).
"""

from __future__ import annotations

import logging
import queue
import threading
from dataclasses import dataclass
from pathlib import Path

from PySide6.QtCore import QThread, Signal

from morsedesk.api import DEFAULT_MODEL_PATH, Decoder, StreamDecoder
from morsedesk.app.worker import Spectrogram
from morsedesk.audio.capture import BLOCK_MS, MAX_BLOCKS, AudioCapture, CaptureError
from morsedesk.constants import SAMPLE_RATE
from morsedesk.lm.store import LMStore

log = logging.getLogger(__name__)

# Mã lỗi của `LiveWorker.failed`; UI dựng thông điệp bằng tr() theo mã, không hiển thị chuỗi gốc.
LIVE_FAILURE_CODES = ("device", "config", "internal")
WATERFALL_COLUMNS_PER_BLOCK = 4      # 100 ms -> 4 cột (40 cột/giây) cho waterfall


@dataclass(frozen=True)
class LiveRequest:
    device: int | None
    profile_id: str
    use_lm: bool
    fallback: bool = True


class LiveWorker(QThread):
    """Vòng ngoài chặn ở hàng đợi lệnh; vòng trong chặn ở hàng đợi khối của `AudioCapture`."""

    words_ready = Signal(object)     # list[Word] mới ổn định (thời gian tuyệt đối từ lúc bắt đầu)
    stats = Signal(object)           # DecodeResult của cửa sổ vừa giải mã
    spectrum = Signal(object)        # Spectrogram của khối vừa thu (đã gộp cột)
    dropped = Signal(int)            # tổng số khối bị bỏ từ lúc bắt đầu
    listening = Signal(bool)
    failed = Signal(str, str)        # code thuộc LIVE_FAILURE_CODES, detail gốc

    def __init__(self, engine: str, model_path: Path | None = None, lm_dir: Path | None = None,
                 block_ms: int = BLOCK_MS, max_blocks: int = MAX_BLOCKS, parent=None) -> None:
        super().__init__(parent)
        self.engine = engine
        self.model_path = Path(model_path) if model_path else None
        self.lm_dir = Path(lm_dir) if lm_dir else None
        self.block_ms = int(block_ms)
        self.max_blocks = int(max_blocks)
        self._commands: queue.Queue = queue.Queue()
        self._stop_listen = threading.Event()
        self._listening = threading.Event()
        self._capture: AudioCapture | None = None
        self._acoustic = None
        self._store: LMStore | None = None

    # ---- gọi từ UI thread ----
    def start_listening(self, req: LiveRequest) -> None:
        self._commands.put(req)

    def stop_listening(self) -> None:
        self._stop_listen.set()

    def is_listening(self) -> bool:
        return self._listening.is_set()

    def pending_blocks(self) -> int:
        cap = self._capture
        return cap.pending if cap is not None else 0

    def stop_async(self) -> None:
        """Xin thread dừng hẳn nhưng KHÔNG chờ (gọi được từ `closeEvent`)."""
        self._stop_listen.set()
        self._commands.put(None)

    def stop(self) -> None:
        self.stop_async()
        self.wait(10000)

    # ---- chạy trong thread ----
    def _decoder(self, req: LiveRequest) -> Decoder:
        if self._store is None:
            self._store = LMStore(self.lm_dir)
        if self.engine == "onnx" and self._acoustic is None:
            from morsedesk.model.onnx_runner import OnnxAcousticModel

            self._acoustic = OnnxAcousticModel(self.model_path or DEFAULT_MODEL_PATH)
        return Decoder(
            req.profile_id, engine=self.engine, model_path=self.model_path, lm=req.use_lm,
            fallback=req.fallback, lm_store=self._store, acoustic_model=self._acoustic,
        )

    def _listen(self, req: LiveRequest) -> None:
        self._stop_listen.clear()
        try:
            stream = StreamDecoder(req.profile_id, decoder=self._decoder(req))
        except (FileNotFoundError, ValueError, NotImplementedError) as e:
            log.warning("live config error: %s", e)
            self.failed.emit("config", str(e))
            return
        capture = AudioCapture(device=req.device, block_ms=self.block_ms,
                               sample_rate=SAMPLE_RATE, max_blocks=self.max_blocks)
        try:
            capture.start()
        except CaptureError as e:
            log.warning("live device error: %s", e)
            self.failed.emit("device", str(e))
            return
        self._capture = capture
        self._listening.set()
        self.listening.emit(True)
        ref_power: float | None = None
        last_dropped = 0
        last_result = None
        try:
            while not self._stop_listen.is_set():
                block = capture.get(timeout=0.2)
                if block is None:
                    continue
                spec = Spectrogram.from_audio(block, SAMPLE_RATE,
                                              max_columns=WATERFALL_COLUMNS_PER_BLOCK,
                                              ref_power=ref_power)
                ref_power = spec.peak_power if ref_power is None else max(ref_power, spec.peak_power)
                self.spectrum.emit(spec)
                words = stream.feed(block)
                if words:
                    self.words_ready.emit(words)
                if stream.last_result is not last_result:     # ~1 lần/giây, không phải mỗi khối
                    last_result = stream.last_result
                    self.stats.emit(last_result)
                if capture.dropped != last_dropped:
                    last_dropped = capture.dropped
                    self.dropped.emit(last_dropped)
            tail = stream.flush()
            if tail:
                self.words_ready.emit(tail)
        except Exception as e:      # không để thread chết vì một khối lạ
            log.exception("live decode failed")
            self.failed.emit("internal", f"{type(e).__name__}: {e}")
        finally:
            capture.stop()
            self._capture = None
            self._listening.clear()
            self.listening.emit(False)

    def run(self) -> None:
        while True:
            req = self._commands.get()
            if req is None:
                return
            self._listen(req)
```

```python
"""Waterfall cuộn theo thời gian thực: đệm cột cố định, mỗi khối thu đẩy vài cột vào bên phải."""

from __future__ import annotations

import numpy as np
import pyqtgraph as pg
from PySide6.QtCore import QEvent, QRectF

from morsedesk.app.worker import DB_FLOOR, MAX_FREQ_HZ, Spectrogram

WATERFALL_SECONDS = 30.0     # bề rộng cửa sổ nhìn thấy
WATERFALL_COLUMNS = 1200     # 40 cột/giây (4 cột cho mỗi khối 100 ms)


class WaterfallView(pg.PlotWidget):
    """Ảnh phổ cuộn: `append()` đẩy cột mới vào mép phải, cột cũ nhất rơi ra mép trái."""

    def __init__(self, parent=None, seconds: float = WATERFALL_SECONDS,
                 columns: int = WATERFALL_COLUMNS) -> None:
        super().__init__(parent, background="k")
        # PlotWidget.__init__ gán self.clear = plotItem.clear làm thuộc tính instance (che mất
        # phương thức clear() của lớp con); bỏ nó đi để clear() bên dưới có hiệu lực.
        self.__dict__.pop("clear", None)
        self.seconds = float(seconds)
        self.n_columns = int(columns)
        self._buf: np.ndarray | None = None
        self.retranslate()
        self.setYRange(0, MAX_FREQ_HZ, padding=0)
        self.setXRange(-self.seconds, 0.0, padding=0)
        self.setMouseEnabled(x=False, y=False)
        self.image = pg.ImageItem()
        self.image.setLookupTable(pg.colormap.get("viridis").getLookupTable())
        self.addItem(self.image)
        self.tone_line = pg.InfiniteLine(angle=0, pen=pg.mkPen("#ff5555", width=2))
        self.tone_line.setVisible(False)
        self.addItem(self.tone_line)

    def retranslate(self) -> None:
        self.setLabel("bottom", self.tr("s"))
        self.setLabel("left", self.tr("Hz"))

    def changeEvent(self, event: QEvent) -> None:  # noqa: N802 (tên Qt)
        if event.type() == QEvent.Type.LanguageChange:
            self.retranslate()
        super().changeEvent(event)

    def columns(self) -> int:
        """Số cột đang giữ trong đệm (0 khi chưa có khối nào)."""
        return 0 if self._buf is None else int(self._buf.shape[1])

    def append(self, spec: Spectrogram) -> None:
        """Nối phổ của một khối vào mép phải; khối rỗng bị bỏ qua."""
        if spec is None or spec.power_db.size == 0:
            return
        cols = spec.power_db[:, -self.n_columns:]
        if self._buf is None or self._buf.shape[0] != cols.shape[0]:
            self._buf = np.full((cols.shape[0], self.n_columns), DB_FLOOR, dtype=np.float32)
            f_end = float(spec.freqs_hz[-1])
            self.image.setRect(QRectF(-self.seconds, 0.0, self.seconds, f_end))
        k = cols.shape[1]
        self._buf = np.roll(self._buf, -k, axis=1)
        self._buf[:, -k:] = cols
        self.image.setImage(np.ascontiguousarray(self._buf.T), autoLevels=False,
                            levels=(DB_FLOOR, 0.0))

    def set_tone(self, hz: float | None) -> None:
        """Vạch ngang ở tần số tone đang bám; `None` ẩn vạch."""
        if hz is None:
            self.tone_line.setVisible(False)
```

```python
def run_chunked(model: AcousticModel, feats: np.ndarray, window: int = WINDOW_FRAMES,
                step: int = STEP_FRAMES, progress: ProgressFn | None = None,
                cancel: CancelFn | None = None) -> np.ndarray:
    """Đặc trưng đã tính cho cả clip (chế độ 'clip')."""
    n = feats.shape[0]
    spans = frame_windows(n, window, step)
    pieces = []
    for i, (s, e) in enumerate(spans):
        if cancel is not None and cancel():
            raise DecodeCancelled(f"hủy sau {i}/{len(spans)} cửa sổ")
        pieces.append((s, e, model(feats[s:e])))
        if progress is not None:
            progress(i + 1, len(spans))
    return _merge_pieces(pieces, n, int(model.time_stride), len(model.meta["vocab"]))


def run_chunked_windows(model: AcousticModel, windows: list[Window], n_frames: int,
                        progress: ProgressFn | None = None,
                        cancel: CancelFn | None = None) -> np.ndarray:
    """Chế độ 'window' với đặc trưng đã tính sẵn (tránh tính lại khi bên gọi đã có cửa sổ)."""
    pieces = []
    for i, w in enumerate(windows):
        if cancel is not None and cancel():
            raise DecodeCancelled(f"hủy sau {i}/{len(windows)} cửa sổ")
        pieces.append((w.start_frame, w.end_frame, model(w.feats)))
        if progress is not None:
            progress(i + 1, len(windows))
    return _merge_pieces(pieces, n_frames, int(model.time_stride), len(model.meta["vocab"]))


def run_chunked_audio(model: AcousticModel, x: np.ndarray, f0_hint: float | None = None,
                      window: int = WINDOW_FRAMES, step: int = STEP_FRAMES,
                      progress: ProgressFn | None = None,
                      cancel: CancelFn | None = None) -> tuple[np.ndarray, int]:
    """Chế độ 'window': đặc trưng và tone tính riêng từng cửa sổ. Trả (log-prob gộp, số khung đầu vào)."""
    n_frames = len(x) // HOP
    windows = compute_features_windowed(x, f0_hint, window_frames=window, step_frames=step)
    return run_chunked_windows(model, windows, n_frames, progress, cancel), n_frames
```

```python
def iter_audio_blocks(path: str | Path, block_s: float = 4.0,
                      target_sr: int = SAMPLE_RATE) -> Iterator[np.ndarray]:
    """Đọc file theo khối `block_s` giây, trả mono float32 ở `target_sr`.

    Khác `load_audio` ở hai điểm, đều có chủ ý: (1) không bao giờ giữ cả file trong RAM, nhờ
    đó file 1 giờ vẫn chạy với bộ nhớ hằng số (CP-14); (2) KHÔNG chuẩn hóa biên độ, vì chuẩn
    hóa theo từng khối sẽ bơm mức nền của những khối im lặng lên đúng bằng khối có tín hiệu.
    Mức tín hiệu do AGC của `dsp.frontend` lo.
    """
    path = Path(path)
    if not path.exists():
        raise FileNotFoundError(f"không tìm thấy file: {path}")
    try:
        f = sf.SoundFile(str(path))
    except sf.LibsndfileError as e:
        raise UnsupportedAudio(
            f"không đọc được {path.name}: {e}. Định dạng hỗ trợ: {', '.join(SUPPORTED_EXT)}"
        ) from e
    with f:
        frames = max(1, int(block_s * f.samplerate))
        for block in f.blocks(blocksize=frames, dtype="float32", always_2d=True):
            if not len(block):
                continue
            mono = block.mean(axis=1) if block.shape[1] > 1 else block[:, 0]
            mono = mono.astype(np.float32) - np.float32(np.mean(mono))   # bỏ DC như load_audio
            yield resample(mono, int(f.samplerate), target_sr)
```

```python
def concat_spectrograms(parts: list[tuple[float, Spectrogram]]) -> Spectrogram:
    """Ghép các mảnh `(thời điểm bắt đầu, Spectrogram)` thành một spectrogram liên tục."""
    usable = [(t, s) for t, s in parts if s.power_db.size]
    if not usable:
        return Spectrogram(np.zeros((0, 0), np.float32), np.zeros(0), np.zeros(0), 0.0)
    power = np.concatenate([s.power_db for _, s in usable], axis=1)
    times = np.concatenate([s.times_s + t for t, s in usable])
    return Spectrogram(power, usable[0][1].freqs_hz, times,
                       max(s.peak_power for _, s in usable))
```

```python
    def _handle(self, req: DecodeRequest) -> None:
        """Giải mã một yêu cầu; lỗi đã phân loại được ném dưới dạng `DecodeFailure`."""
        path = Path(req.path)
        if not path.exists():
            raise DecodeFailure("not_found", f"file not found: {path}")
        try:
            _, _, seconds = audio_info(path)
        except Exception as e:  # soundfile không đọc được (định dạng lạ, file hỏng)
            raise DecodeFailure("unsupported", f"cannot read {path.name}: {e}") from e
        if seconds > MAX_SECONDS:
            raise DecodeFailure("too_long", f"{seconds / 60:.1f}")   # detail: số phút đã format
        decoder = self._decoder(req)
        if seconds > STREAM_AFTER_S:
            result, spec = self._decode_streaming(req, decoder, seconds)
        else:
            result, spec = self._decode_once(req, decoder)
        self.result_ready.emit(result, spec, str(path))

    def _decode_once(self, req: DecodeRequest, decoder: Decoder) -> tuple[DecodeResult, Spectrogram]:
        """File ngắn: nạp cả file, giải mã một lần. Tiến trình theo cửa sổ của `chunker`."""
        try:
            x, sr = load_audio(req.path)
        except FileNotFoundError as e:                # file biến mất sau lần kiểm exists()
            raise DecodeFailure("not_found", str(e)) from e
        except UnsupportedAudio as e:
            raise DecodeFailure("unsupported", str(e)) from e
        self.progress.emit(0, 1)
        result = decoder.decode_array(x, sr, progress=self._on_progress, cancel=self._cancel.is_set)
        self.progress.emit(1, 1)
        return result, Spectrogram.from_audio(x, SAMPLE_RATE)

    def _decode_streaming(self, req: DecodeRequest, decoder: Decoder,
                          seconds: float) -> tuple[DecodeResult, Spectrogram]:
        """File dài (tới 1 giờ): đọc theo khối, giải mã theo luồng, bộ nhớ hằng số (CP-14)."""
        stream = StreamDecoder(req.profile_id, decoder=decoder)
        total = max(1, int(math.ceil(seconds / STREAM_BLOCK_S)))
        parts: list[tuple[float, Spectrogram]] = []
        ref_power: float | None = None
        t = 0.0
        self.progress.emit(0, total)
        try:
            for i, block in enumerate(iter_audio_blocks(req.path, STREAM_BLOCK_S)):
                if self._cancel.is_set():
                    raise DecodeCancelled(f"hủy sau {i}/{total} khối")
                stream.feed(block)
                spec = Spectrogram.from_audio(block, SAMPLE_RATE,
                                              max_columns=STREAM_COLUMNS_PER_BLOCK,
                                              ref_power=ref_power)
                ref_power = spec.peak_power if ref_power is None else max(ref_power, spec.peak_power)
                parts.append((t, spec))
                t += len(block) / SAMPLE_RATE
                self.progress.emit(min(i + 1, total), total)
        except UnsupportedAudio as e:
            raise DecodeFailure("unsupported", str(e)) from e
        stream.flush()
        self.progress.emit(total, total)
        return stream.result(), concat_spectrograms(parts)

    def _on_progress(self, done: int, total: int) -> None:
        self.progress.emit(int(done), int(total))
```

Thay vòng `run()` để đặt lại cờ hủy và bắt `DecodeCancelled`:

```python
    def run(self) -> None:
        while True:
            req = self._queue.get()
            if req is None:
                return
            self._cancel.clear()
            self._busy = True
            self.busy.emit(True)
            try:
                self._handle(req)
            except DecodeCancelled as e:
                log.info("decode cancelled: %s", e)
                self.cancelled.emit(str(Path(req.path)))
            except DecodeFailure as e:
                log.warning("decode failed (%s): %s", e.code, e.detail)
                self.failed.emit(e.code, e.detail, str(Path(req.path)))
            except (FileNotFoundError, ValueError, NotImplementedError) as e:
                # Cài đặt sai: thiếu file LM/model, engine chưa hỗ trợ, tổ hợp profile/LM không hợp lệ.
                log.warning("decode config error: %s", e)
                self.failed.emit("config", str(e), str(Path(req.path)))
            except Exception as e:  # không để worker chết vì một file lạ
                log.exception("decode failed")
                self.failed.emit("internal", f"{type(e).__name__}: {e}", str(Path(req.path)))
            finally:
                self._busy = False
                self.busy.emit(False)
```

```python
@dataclass(frozen=True)
class Mp3Support:
    """Khả năng ghi MP3 dò được lúc khởi động; trang Cài đặt hiển thị `method`."""

    libsndfile: bool
    ffmpeg: str | None

    @property
    def available(self) -> bool:
        return self.libsndfile or self.ffmpeg is not None

    @property
    def method(self) -> str:
        if self.libsndfile:
            return "libsndfile"
        return "ffmpeg" if self.ffmpeg else "none"


def mp3_support() -> Mp3Support:
    """Dò một lần: libsndfile có LAME không, và có `ffmpeg` trên PATH không."""
    try:
        lame = bool(sf.check_format("MP3"))
    except Exception:       # bản libsndfile quá cũ không biết tên format "MP3"
        lame = False
    return Mp3Support(lame, shutil.which("ffmpeg"))


def write_mp3(path: str | Path, x: np.ndarray, sample_rate: int,
              support: Mp3Support | None = None) -> None:
    """Ghi MP3 theo ba nhánh của spec §4.3: libsndfile → ffmpeg ngoài → báo rõ là không được."""
    s = support if support is not None else mp3_support()
    path = Path(path)
    if s.libsndfile:
        sf.write(str(path), x, sample_rate, format="MP3")
        return
    if s.ffmpeg:
        with tempfile.TemporaryDirectory() as d:
            wav = Path(d) / "in.wav"
            sf.write(str(wav), x, sample_rate, format="WAV", subtype="PCM_16")
            proc = subprocess.run(
                [s.ffmpeg, "-y", "-loglevel", "error", "-i", str(wav),
                 "-codec:a", "libmp3lame", "-q:a", MP3_QUALITY, str(path)],
                capture_output=True, text=True,
            )
        if proc.returncode != 0:
            raise Mp3Unavailable(f"ffmpeg lỗi ({proc.returncode}): {(proc.stderr or '').strip()[:200]}")
        return
    raise Mp3Unavailable(
        "không xuất được MP3: libsndfile không có LAME và không tìm thấy ffmpeg trên PATH"
    )
```

```python
def save_audio(path: str | Path, x: np.ndarray, sample_rate: int) -> None:
    """Ghi audio theo phần mở rộng của `path`, biên độ clip về [-1, 1].

    Format lấy từ `EXT_FORMAT` chứ không để libsndfile đoán theo đuôi. Chỉ ép `PCM_16` cho
    container nhận subtype PCM (WAV, AIFF, FLAC); OGG không nhận PCM_16 — libsndfile ném
    `ValueError: Invalid combination of format, subtype and endian` — nên để mặc định (Vorbis).
    """
    path = Path(path)
    fmt = EXT_FORMAT.get(path.suffix.lower())
    if fmt is None:
        raise UnsupportedAudio(
            f"không ghi được {path.name}: đuôi {path.suffix or '(trống)'} không hỗ trợ. "
            f"Định dạng hỗ trợ: {', '.join(SUPPORTED_EXT)}"
        )
    data = np.clip(x, -1.0, 1.0)
    if fmt == "MP3":
        write_mp3(path, data, sample_rate)
        return
    subtype = "PCM_16" if sf.check_format(fmt, "PCM_16") else None
    sf.write(str(path), data, sample_rate, format=fmt, subtype=subtype)
```

```python
    @property
    def default_profile(self) -> str:
        """Profile mặc định của trang Giải mã; giá trị lạ trong QSettings -> 'auto'."""

        v = self._s.value(key, None)

        except (TypeError, ValueError):

        return i if i >= 0 else None

    @property
    def input_device(self) -> int | None:

        self._s.sync()

    @property
    def output_device(self) -> int | None:
        return self._device("audio/output_device")

    @output_device.setter
    def output_device(self, index: int | None) -> None:
        self._s.setValue("audio/output_device", -1 if index is None else int(index))
        self._s.sync()

    @property
    def export_dir(self) -> str:
        """Thư mục xuất mặc định; chưa đặt thì dùng thư mục gần nhất."""

        return v if isinstance(v, str) and v else self.last_dir

    @export_dir.setter
    def export_dir(self, path: str) -> None:
        self._s.setValue("paths/export_dir", path)
        self._s.sync()

    @property
    def dsp_fallback(self) -> bool:
        """Bật fallback DSP khi tin cậy thấp; giá trị hỏng -> True (mặc định của lõi)."""
        v = self._s.value("decode/fallback", None)
        if isinstance(v, bool):
            return v
        if isinstance(v, str) and v.lower() in ("true", "1", "false", "0"):
```

```python
"""Trang Cài đặt (tab thứ ba): profile mặc định, thiết bị thu/phát, thư mục xuất, fallback DSP,
khả năng xuất MP3 dò được lúc khởi động. Mọi thay đổi ghi thẳng vào `AppSettings` (QSettings)."""

from __future__ import annotations

from PySide6.QtCore import QEvent, Signal
from PySide6.QtWidgets import (
    QCheckBox,
    QComboBox,
    QFileDialog,
    QFormLayout,
    QHBoxLayout,
    QLabel,
    QLineEdit,
    QPushButton,
    QVBoxLayout,
    QWidget,
)

from morsedesk.app.settings import PROFILE_IDS, AppSettings
from morsedesk.app.startup import Capabilities
from morsedesk.audio.capture import CaptureError, list_input_devices, list_output_devices


class SettingsPage(QWidget):
    """Năm mục của spec §4.2 cộng dòng nhắc ngôn ngữ giao diện (combo vẫn ở thanh công cụ)."""

    changed = Signal()

    def __init__(self, settings: AppSettings, capabilities: Capabilities, parent=None) -> None:
        super().__init__(parent)
        self.settings = settings
        self.capabilities = capabilities

        self.profile_combo = QComboBox(self)
        self.profile_combo.setSizeAdjustPolicy(QComboBox.SizeAdjustPolicy.AdjustToContents)
        for pid in PROFILE_IDS:
            self.profile_combo.addItem(pid, pid)
        self.input_combo = QComboBox(self)
        self.output_combo = QComboBox(self)
        for combo in (self.input_combo, self.output_combo):
            combo.setSizeAdjustPolicy(QComboBox.SizeAdjustPolicy.AdjustToContents)
        self.export_edit = QLineEdit(self)
        self.export_edit.setReadOnly(True)
        self.export_button = QPushButton(self)
        export_row = QWidget(self)
        export_lay = QHBoxLayout(export_row)
        export_lay.setContentsMargins(0, 0, 0, 0)
        export_lay.addWidget(self.export_edit, 1)
        export_lay.addWidget(self.export_button, 0)
        self.fallback_check = QCheckBox(self)
        self.mp3_value = QLabel(self)
        self.language_hint = QLabel(self)
        self.language_hint.setStyleSheet("color: gray;")

        self.form = QFormLayout()
        self.form_labels = [QLabel(self) for _ in range(5)]
        rows = (self.profile_combo, self.input_combo, self.output_combo, export_row, self.mp3_value)
        for label, w in zip(self.form_labels, rows, strict=True):
            self.form.addRow(label, w)
        lay = QVBoxLayout(self)
        lay.addLayout(self.form)
        lay.addWidget(self.fallback_check)
        lay.addWidget(self.language_hint)
        lay.addStretch(1)

        self.refresh_devices()
        self._load()
        self.profile_combo.currentIndexChanged.connect(self._on_profile)
        self.input_combo.currentIndexChanged.connect(self._on_input)
        self.output_combo.currentIndexChanged.connect(self._on_output)
        self.fallback_check.toggled.connect(self._on_fallback)
        self.export_button.clicked.connect(self._choose_export_dir)
        self.retranslate()

    # ---- nạp và lưu ----
    def refresh_devices(self) -> None:
        """Nạp lại hai combo thiết bị; máy không có PortAudio thì chỉ còn mục 'theo hệ thống'."""
        for combo, lister in ((self.input_combo, list_input_devices),
                              (self.output_combo, list_output_devices)):
            combo.blockSignals(True)
            combo.clear()
            combo.addItem("", None)              # văn bản đặt trong retranslate()
            try:
                for d in lister():
                    combo.addItem(d.name, d.index)
            except CaptureError:
                pass                              # không có thiết bị: vẫn dùng được phần còn lại
            combo.blockSignals(False)

    def _load(self) -> None:
        """Đặt trạng thái widget theo `AppSettings` mà không phát `changed`."""
        for combo, value in ((self.profile_combo, self.settings.default_profile),
                             (self.input_combo, self.settings.input_device),
                             (self.output_combo, self.settings.output_device)):
            combo.blockSignals(True)
            index = combo.findData(value)
            combo.setCurrentIndex(index if index >= 0 else 0)
            combo.blockSignals(False)
        self.fallback_check.blockSignals(True)
        self.fallback_check.setChecked(self.settings.dsp_fallback)
        self.fallback_check.blockSignals(False)
        self.export_edit.setText(self.settings.export_dir)

    def _on_profile(self, _index: int) -> None:
        self.settings.default_profile = str(self.profile_combo.currentData())
        self.changed.emit()

    def _on_input(self, _index: int) -> None:
        self.settings.input_device = self.input_combo.currentData()
        self.changed.emit()

    def _on_output(self, _index: int) -> None:
        self.settings.output_device = self.output_combo.currentData()
        self.changed.emit()

    def _on_fallback(self, flag: bool) -> None:
        self.settings.dsp_fallback = bool(flag)
        self.changed.emit()

    def set_export_dir(self, path: str) -> None:
        """Đặt thư mục xuất (API công khai để test và mã khác gọi thẳng, không qua hộp thoại)."""
        self.settings.export_dir = path
        self.export_edit.setText(path)
        self.changed.emit()

    def _choose_export_dir(self) -> None:
        path = QFileDialog.getExistingDirectory(
            self, self.tr("Choose export folder"), self.settings.export_dir
        )
        if path:
            self.set_export_dir(path)

    # ---- i18n ----
    def mp3_text(self) -> str:
        """Câu mô tả khả năng xuất MP3 dò được lúc khởi động (CP-06)."""
        method = self.capabilities.mp3.method
        if method == "libsndfile":
            return self.tr("available through libsndfile")
        if method == "ffmpeg":
            return self.tr("available through the external ffmpeg")
        return self.tr("not available (no LAME in libsndfile, no ffmpeg on PATH)")

    def retranslate(self) -> None:
        names = {
            "auto": self.tr("Auto"),
            "en": self.tr("English"),
            "ru": self.tr("Russian"),
            "kk-cyrl-ru": self.tr("Kazakh"),
        }
        for i in range(self.profile_combo.count()):
            self.profile_combo.setItemText(i, names[self.profile_combo.itemData(i)])
        for combo in (self.input_combo, self.output_combo):
            combo.setItemText(0, self.tr("System default"))
        for label, text in zip(
            self.form_labels,
```

```python
class Player:
    """Phát thử qua sounddevice (không chặn); hỏng thì trả `(mã, chi tiết)` chứ không ném."""

    def play(self, x: np.ndarray, sample_rate: int,
             device: int | None = None) -> tuple[str, str] | None:
        try:
            import sounddevice as sd
        except Exception as e:      # ImportError, hoặc OSError khi máy thiếu libportaudio2
            return ("no_backend", f"{type(e).__name__}: {e}")
        try:
            sd.stop()   # phát lại từ đầu nếu bấm Nghe thử lần nữa khi đang phát
            sd.play(np.asarray(x, dtype=np.float32), sample_rate, device=device)
        except Exception as e:      # PortAudioError khi thiết bị bận/biến mất
            return ("play_failed", f"{type(e).__name__}: {e}")
        return None
```

```python
def log_dir() -> Path:
    """Thư mục log: `<AppDataLocation>/logs`.

    CP-15 chốt theo Qt, không ép `~/.local/share/morsedesk/logs` của spec gốc §7 (spec đã được
    sửa cho khớp). Trên Linux: `~/.local/share/MorseDesk/MorseDesk/logs`; trên Windows:
    `%APPDATA%\\MorseDesk\\MorseDesk\\logs`. `set_app_identity()` phải chạy trước, nếu không
    `AppDataLocation` trả về thư mục gốc và log rơi ra ngoài.
    """
    base = QStandardPaths.writableLocation(QStandardPaths.StandardLocation.AppDataLocation)
    return Path(base) / "logs"


def setup_logging() -> None:
    """Ghi log xoay vòng (1 MB x 3) vào `log_dir()`; rơi về basicConfig nếu không tạo được."""
    directory = log_dir()
    try:
        directory.mkdir(parents=True, exist_ok=True)
        handler = RotatingFileHandler(
            directory / "morsedesk.log", maxBytes=1_000_000, backupCount=3, encoding="utf-8"
        )
        handler.setFormatter(logging.Formatter("%(asctime)s %(levelname)s %(name)s: %(message)s"))
        logging.getLogger().addHandler(handler)
        logging.getLogger().setLevel(logging.INFO)
    except OSError:
        logging.basicConfig(level=logging.INFO)
```

## Điều kiện kết thúc kế hoạch này

Đối chiếu với spec tuần 5 mục 7 (chỉ phần thuộc nhánh B và C):

1. `uv run pytest` xanh trên Windows **và** Ubuntu 22.04/24.04; `uv run ruff check morsedesk tests tools` sạch; `uv run python tools/build_i18n.py --check` sạch. → Task 1–9.
2. Chế độ live chạy: mở thiết bị, hiện chữ theo thời gian thực, đếm khối bỏ, dừng sạch. → Task 1, 3, 4.
3. Trang Cài đặt đủ năm mục, lưu và nạp lại đúng. → Task 7.
4. CP-06, CP-07, CP-08, CP-09, CP-10, CP-14, CP-15 đóng, mỗi cái có test. → Task 5, 6, 8 (bảng bằng chứng ở Task 8 Step 12).
5. Toàn bộ chạy trên Ubuntu 22.04 và 24.04. → Task 9.

Ngoài phạm vi kế hoạch này (kế hoạch nhánh A lo): mục 2, 3, 4, 5 của spec §7 — engine `deepcw`,
bộ T2, thang thí nghiệm E6–E10, `pytest -m gate`.

## Tự rà soát (đã chạy khi viết kế hoạch)

**1. Phủ spec.** Mục 4.1 → Task 1 (`capture.py`), Task 2 (`StreamDecoder`), Task 3 (`LiveWorker`,
kèm lý do không dùng `moveToThread`), Task 4 (combo thiết bị, Bắt đầu/Dừng, waterfall, đếm khối bỏ
trên `StatsBar`). Mục 4.2 → Task 7 (năm mục + dòng nhắc ngôn ngữ giao diện, `QSettings` qua
`AppSettings`), phần "khả năng xuất MP3 dò được lúc khởi động" → Task 6 cung cấp, Task 7 hiển thị.
Mục 4.3: CP-08 → Task 5; CP-09 → Task 5; CP-10 → Task 8; CP-06 → Task 6; CP-07 → Task 6; CP-14 →
Task 5; CP-15 → Task 8. Mục 5 → Task 9. Mục 7 điều kiện 1, 6, 7, 8, 9 → bảng ngay trên.

**Hai chỗ spec không biến thành task được, và lý do:**
- CP-05 đòi "`morsedesk-gui` mở thật và giải mã một file trên **máy Linux không phải WSL**". Máy
  phát triển chỉ có Docker và WSL; container không có máy chủ hiển thị thật còn WSLg vẫn là WSL.
  Task 9 làm hết phần làm được và ghi phần còn lại là **còn mở, chuyển tuần 6** (kiểm AppImage trên
  VM sạch cũng đang nằm ở tuần 6, đúng chỗ để đóng nốt).
- Mục 6 của spec (T3 loopback, đóng gói, AppImage, font) thuộc tuần 6, không nằm trong phạm vi
  được giao; Task 9 Step 5 chỉ tiện tay cài 8 thư viện của CP-16 để trả lời trước câu hỏi `xcb`.

**2. Quét chỗ trống.** Không có "TBD", "tương tự Task N", "thêm xử lý lỗi phù hợp". Mọi bước đổi mã
đều kèm mã hoàn chỉnh; mọi bước chạy lệnh đều kèm kết quả mong đợi. Các con số đã đo thật trên máy
này (libsndfile 1.2.2 ghi được MP3; `.aif`/`.oga` ném `TypeError`; `САЛЕМ АЛЕМ` là kết quả CLI của
`testdata/generated/kk.wav`) được ghi thẳng vào task để người thực thi không phải đoán.

**3. Nhất quán kiểu giữa các task.**
- `AudioCapture.get/pending/dropped/start/stop` (Task 1) dùng đúng tên đó ở `LiveWorker._listen`
  (Task 3) và `LiveWorker.pending_blocks` (Task 3, được test Task 3 và Task 4 gọi).
- `StreamDecoder.feed/flush/result/last_result/words/text/reset` (Task 2) dùng đúng tên đó ở
  `LiveWorker` (Task 3) và `DecoderWorker._decode_streaming` (Task 5).
- `Spectrogram(power_db, freqs_hz, times_s, peak_power)` và `from_audio(..., ref_power=)` định
  nghĩa ở Task 3, dùng lại ở Task 4 (`WaterfallView.append`) và Task 5 (`concat_spectrograms`).
- `LiveRequest(device, profile_id, use_lm, fallback)` định nghĩa ở Task 3, dựng ở Task 4
  (`current_live_request`) và nhận `settings.dsp_fallback` ở Task 7.
- `DecodeRequest` thêm trường `fallback` ở Task 7 với giá trị mặc định `True`, nên mọi lời gọi ba
  đối số ở Task 4–5 và trong test cũ vẫn đúng.
- `Player.play` nhận `device=` ở Task 7 (vẫn trả `str | None`) rồi đổi sang trả `tuple[str, str] |
  None` ở Task 8; Task 8 sửa luôn `EncodePage.play` và hai test `Player` cho khớp — không còn chỗ
  nào đọc giá trị trả về theo kiểu cũ.
- `Capabilities`/`Mp3Support` định nghĩa ở Task 6, dùng ở Task 6 (`EncodePage`) và Task 7
  (`SettingsPage.mp3_text`, `MainWindow`).
- `DecodeWarning`/`WARN_*` đổi ở Task 8 chỉ ảnh hưởng nơi **đọc** `warnings`: `decode_page`,
  `cli`, `bench` — cả ba đều được sửa trong chính Task 8. `StreamDecoder` (Task 2) chỉ sao chép
  phần tử và dedup bằng `repr`, nên không phải sửa lại.
- `StatsBar.set_dropped` (Task 4) và `StatsBar.set_progress` (Task 5) là hai phương thức khác nhau,
  không đè lên nhau; cả hai đều được gọi lại từ `retranslate()`.
